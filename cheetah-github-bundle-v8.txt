typedef Item = Map<String, dynamic>;

final db = FirebaseFirestore.instance;

class Api {
  static String tmdbKey = '';
  static String omdbKey = '';

  static Future<void> load() async {
    try {
      final d = await db.collection('config').doc('api').get();
      final m = d.data() ?? {};
      tmdbKey = '${m['tmdbKey'] ?? ''}'.trim();
      omdbKey = '${m['omdbKey'] ?? ''}'.trim();
    } catch (_) {}
  }

  static Future<Map<String, dynamic>?> tmdb(String path,
      [Map<String, String> q = const {}]) async {
    if (tmdbKey.isEmpty) return null;
    try {
      final uri =
          Uri.https('api.themoviedb.org', '/3/$path', {'api_key': tmdbKey, ...q});
      final r = await http.get(uri).timeout(const Duration(seconds: 15));
      if (r.statusCode == 200) {
        return Map<String, dynamic>.from(jsonDecode(r.body) as Map);
      }
    } catch (_) {}
    return null;
  }

  static Future<Map<String, dynamic>?> omdb(String imdbId) async {
    if (omdbKey.isEmpty || imdbId.isEmpty) return null;
    try {
      final uri = Uri.https('www.omdbapi.com', '/', {'apikey': omdbKey, 'i': imdbId});
      final r = await http.get(uri).timeout(const Duration(seconds: 15));
      if (r.statusCode == 200) {
        final j = Map<String, dynamic>.from(jsonDecode(r.body) as Map);
        if (j['Response'] == 'True') return j;
      }
    } catch (_) {}
    return null;
  }
}

class _Col {
  final String name;
  final ValueNotifier<List<Item>> notifier;
  final int Function(Item, Item) cmp;
  _Col(this.name, this.notifier, this.cmp);
}

class Store {
  static final content = ValueNotifier<List<Item>>([]);
  static final slides = ValueNotifier<List<Item>>([]);
  static final ads = ValueNotifier<List<Item>>([]);
  static final servers = ValueNotifier<List<Item>>([]);
  static final popups = ValueNotifier<List<Item>>([]);
  static final notifs = ValueNotifier<List<Item>>([]);

  static final ready = ValueNotifier<bool>(false);

  static final List<StreamSubscription> _subs = [];

  static int _ts(Item m) {
    final t = m['createdAt'];
    return t is Timestamp ? t.millisecondsSinceEpoch : 0;
  }

  static num _n(dynamic v) => v is num ? v : 0;

  static int _newest(Item a, Item b) => _ts(b).compareTo(_ts(a));

  static int _byOrder(Item a, Item b) => _n(a['order']).compareTo(_n(b['order']));

  static final List<_Col> _cols = [
    _Col('content', content, _newest),
    _Col('slides', slides, _byOrder),
    _Col('ads', ads, _newest),
    _Col('servers', servers, _byOrder),
    _Col('popups', popups, _newest),
    _Col('notifications', notifs, _newest),
  ];

  static List<Item> _read(QuerySnapshot<Map<String, dynamic>> s, int Function(Item, Item) cmp) {
    final l = s.docs.map((d) => <String, dynamic>{...d.data(), 'id': d.id}).toList();
    l.sort(cmp);
    return l;
  }

  static void start() {
    if (_subs.isNotEmpty) return;
    for (final c in _cols) {
      _subs.add(db.collection(c.name).snapshots().listen((s) {
        c.notifier.value = _read(s, c.cmp);
        if (c.name == 'content') ready.value = true;
      }, onError: (_) {
        if (c.name == 'content') ready.value = true;
      }));
    }
  }

  static void stop() {
    for (final s in _subs) {
      s.cancel();
    }
    _subs.clear();
    ready.value = false;
  }

  static Future<bool> _fetch(_Col c) async {
    try {
      final s = await db
          .collection(c.name)
          .get(const GetOptions(source: Source.server))
          .timeout(const Duration(seconds: 15));
      c.notifier.value = _read(s, c.cmp);
      if (c.name == 'content') ready.value = true;
      return true;
    } catch (_) {
      return false;
    }
  }

  static Future<bool> refresh() async {
    final r = await Future.wait<bool>([
      for (final c in _cols) _fetch(c),
      Api.load().then((_) => true),
    ]);
    return r.every((e) => e);
  }
}

Future<void> openLink(String url) async {
  if (url.trim().isEmpty) return;
  try {
    final u = Uri.tryParse(url.trim());
    if (u != null) await launchUrl(u, mode: LaunchMode.externalApplication);
  } catch (_) {}
}

void openItem(BuildContext c, Item it) => Navigator.push(
      c,
      PageRouteBuilder(
        transitionDuration: const Duration(milliseconds: 280),
        reverseTransitionDuration: const Duration(milliseconds: 220),
        pageBuilder: (_, __, ___) => DetailScreen(item: it),
        transitionsBuilder: (_, a, __, child) => FadeTransition(
          opacity: CurvedAnimation(parent: a, curve: Curves.easeOut),
          child: SlideTransition(
            position: Tween(begin: const Offset(0, 0.04), end: Offset.zero)
                .animate(CurvedAnimation(parent: a, curve: Curves.easeOutCubic)),
            child: child,
          ),
        ),
      ),
    );

class Pull extends StatelessWidget {
  final Widget child;
  const Pull({super.key, required this.child});

  Future<void> _go(BuildContext c) async {
    HapticFeedback.lightImpact();
    final ok = await Store.refresh();
    if (!c.mounted) return;
    final m = ScaffoldMessenger.of(c);
    m.hideCurrentSnackBar();
    m.showSnackBar(SnackBar(
      behavior: SnackBarBehavior.floating,
      duration: const Duration(seconds: 2),
      backgroundColor: kCard2,
      shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(14)),
      content: Row(children: [
        Icon(ok ? Icons.check_circle_rounded : Icons.wifi_off_rounded,
            size: 18, color: ok ? Colors.greenAccent : Colors.orangeAccent),
        const SizedBox(width: 10),
        Text(ok ? 'Up to date' : 'Couldn\'t reach the server'),
      ]),
    ));
  }

  @override
  Widget build(BuildContext context) {
    return RefreshIndicator(
      color: kRed,
      backgroundColor: kCard2,
      displacement: 48,
      onRefresh: () => _go(context),
      child: child,
    );
  }
}

class Skeleton extends StatefulWidget {
  final double? width;
  final double height;
  final double radius;
  const Skeleton({super.key, this.width, required this.height, this.radius = 12});

  @override
  State<Skeleton> createState() => _SkeletonState();
}

class _SkeletonState extends State<Skeleton> with SingleTickerProviderStateMixin {
  late final AnimationController _c =
      AnimationController(vsync: this, duration: const Duration(milliseconds: 1300))..repeat();

  @override
  void dispose() {
    _c.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return AnimatedBuilder(
      animation: _c,
      builder: (_, __) {
        final t = _c.value;
        return Container(
          width: widget.width,
          height: widget.height,
          decoration: BoxDecoration(
            borderRadius: BorderRadius.circular(widget.radius),
            gradient: LinearGradient(
              begin: Alignment(-1.5 + 3 * t, 0),
              end: Alignment(-0.5 + 3 * t, 0),
              colors: const [kCard, kCard2, kCard],
            ),
          ),
        );
      },
    );
  }
}

class _FadeIn extends StatelessWidget {
  final int index;
  final Widget child;
  const _FadeIn({required this.index, required this.child});

  @override
  Widget build(BuildContext context) {
    return TweenAnimationBuilder<double>(
      tween: Tween(begin: 0, end: 1),
      duration: Duration(milliseconds: 280 + (index % 9) * 45),
      curve: Curves.easeOutCubic,
      builder: (_, v, c) => Opacity(
        opacity: v,
        child: Transform.translate(offset: Offset(0, 14 * (1 - v)), child: c),
      ),
      child: child,
    );
  }
}
