from http.server import BaseHTTPRequestHandler
from urllib.parse import urlparse, parse_qs
import json
import yt_dlp


class handler(BaseHTTPRequestHandler):
    def _send(self, code, data):
        self.send_response(code)
        self.send_header("Content-Type", "application/json")
        self.send_header("Access-Control-Allow-Origin", "*")
        self.end_headers()
        self.wfile.write(json.dumps(data).encode())

    def do_GET(self):
        q = parse_qs(urlparse(self.path).query)
        url = (q.get("url") or [""])[0]
        if not url.startswith("http"):
            return self._send(400, {"error": "valid ?url= required"})
        opts = {"quiet": True, "skip_download": True, "noplaylist": True,
                "format": "best[ext=mp4]/best"}
        try:
            with yt_dlp.YoutubeDL(opts) as ydl:
                info = ydl.extract_info(url, download=False)
            link = info.get("url")
            if not link and info.get("formats"):
                link = info["formats"][-1]["url"]
            kind = "hls" if ".m3u8" in (link or "") else "mp4"
            self._send(200, {"url": link, "type": kind, "title": info.get("title")})
        except Exception as e:
            self._send(500, {"error": str(e)[:300]})

import 'dart:convert';
import 'package:cloud_firestore/cloud_firestore.dart';
import 'package:http/http.dart' as http;

// তোমার Vercel লিংক এখানে বসাও
const kApiBase = 'https://cheetah-bd.vercel.app/api/get-link';

class Server {
  final String name, url;
  Server(this.name, this.url);
}

class Content {
  final String id, title, type, poster, backdrop, overview, category, trailer, year;
  final double rating;
  final List<Server> servers;
  Content.fromDoc(DocumentSnapshot d)
      : id = d.id,
        title = (d['title'] ?? '').toString(),
        type = (d.data() as Map)['type']?.toString() ?? 'movie',
        poster = (d.data() as Map)['poster']?.toString() ?? '',
        backdrop = (d.data() as Map)['backdrop']?.toString() ?? '',
        overview = (d.data() as Map)['overview']?.toString() ?? '',
        category = (d.data() as Map)['category']?.toString() ?? 'Latest',
        trailer = (d.data() as Map)['trailer']?.toString() ?? '',
        year = (d.data() as Map)['year']?.toString() ?? '',
        rating = ((d.data() as Map)['rating'] ?? 0).toDouble(),
        servers = (((d.data() as Map)['servers'] ?? []) as List)
            .map((e) => Server(e['name'].toString(), e['url'].toString()))
            .toList();
}

class Api {
  static Stream<List<Content>> content() => FirebaseFirestore.instance
      .collection('content')
      .snapshots()
      .map((s) => s.docs.map(Content.fromDoc).toList());

  /// এমবেড/থার্ড-পার্টি লিংক -> ডাইরেক্ট mp4/m3u8
  static Future<String?> resolve(String url) async {
    if (url.contains('.mp4') || url.contains('.m3u8')) return url;
    try {
      final r = await http
          .get(Uri.parse('$kApiBase?url=${Uri.encodeComponent(url)}'))
          .timeout(const Duration(seconds: 30));
      if (r.statusCode != 200) return null;
      return jsonDecode(r.body)['url'] as String?;
    } catch (_) {
      return null;
    }
  }
}

import 'package:firebase_auth/firebase_auth.dart';
import 'package:flutter/material.dart';

class LoginPage extends StatefulWidget {
  const LoginPage({super.key});
  @override
  State<LoginPage> createState() => _LoginPageState();
}

class _LoginPageState extends State<LoginPage> {
  final email = TextEditingController(), pass = TextEditingController();
  bool signup = false, busy = false;

  Future<void> submit() async {
    setState(() => busy = true);
    try {
      final a = FirebaseAuth.instance;
      signup
          ? await a.createUserWithEmailAndPassword(email: email.text.trim(), password: pass.text)
          : await a.signInWithEmailAndPassword(email: email.text.trim(), password: pass.text);
    } on FirebaseAuthException catch (e) {
      if (mounted) ScaffoldMessenger.of(context).showSnackBar(SnackBar(content: Text(e.message ?? 'Error')));
    }
    if (mounted) setState(() => busy = false);
  }

  @override
  Widget build(BuildContext context) {
    final c = Theme.of(context).colorScheme.primary;
    return Scaffold(
      body: Container(
        decoration: const BoxDecoration(
          gradient: LinearGradient(begin: Alignment.topCenter, end: Alignment.bottomCenter,
              colors: [Color(0xFF2A1405), Color(0xFF0B0B10)]),
        ),
        child: Center(
          child: SingleChildScrollView(
            padding: const EdgeInsets.all(28),
            child: Column(children: [
              Icon(Icons.bolt_rounded, size: 64, color: c),
              const SizedBox(height: 8),
              const Text('CHEETAH BD', style: TextStyle(fontSize: 28, fontWeight: FontWeight.w800, letterSpacing: 3)),
              const SizedBox(height: 32),
              TextField(controller: email, keyboardType: TextInputType.emailAddress, decoration: _d('Email', Icons.mail_outline)),
              const SizedBox(height: 14),
              TextField(controller: pass, obscureText: true, decoration: _d('Password', Icons.lock_outline)),
              const SizedBox(height: 22),
              SizedBox(
                width: double.infinity, height: 50,
                child: FilledButton(
                  onPressed: busy ? null : submit,
                  style: FilledButton.styleFrom(shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(14))),
                  child: busy ? const SizedBox(height: 20, width: 20, child: CircularProgressIndicator(strokeWidth: 2)) : Text(signup ? 'Create account' : 'Login'),
                ),
              ),
              TextButton(onPressed: () => setState(() => signup = !signup),
                  child: Text(signup ? 'Already have an account? Login' : 'New here? Sign up')),
            ]),
          ),
        ),
      ),
    );
  }

  InputDecoration _d(String h, IconData i) => InputDecoration(
        hintText: h, prefixIcon: Icon(i), filled: true, fillColor: Colors.white10,
        border: OutlineInputBorder(borderRadius: BorderRadius.circular(14), borderSide: BorderSide.none),
      );
}

class DetailPage extends StatelessWidget {
  final Content c;
  const DetailPage({super.key, required this.c});

  Future<void> _go(BuildContext ctx, Server s, {required bool download}) async {
    showDialog(context: ctx, barrierDismissible: false,
        builder: (_) => const Center(child: CircularProgressIndicator()));
    final link = await Api.resolve(s.url); // Vercel API call
    if (!ctx.mounted) return;
    Navigator.pop(ctx);
    if (link == null) {
      ScaffoldMessenger.of(ctx).showSnackBar(const SnackBar(content: Text('Link পাওয়া যায়নি, অন্য সার্ভার ট্রাই করো')));
      return;
    }
    if (download) {
      await launchUrl(Uri.parse(link), mode: LaunchMode.externalApplication);
    } else {
      Navigator.push(ctx, MaterialPageRoute(builder: (_) => PlayerView(url: link, title: c.title)));
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: CustomScrollView(slivers: [
        SliverAppBar(
          expandedHeight: 320, pinned: true,
          flexibleSpace: FlexibleSpaceBar(
            background: Stack(fit: StackFit.expand, children: [
              CachedNetworkImage(imageUrl: c.backdrop.isNotEmpty ? c.backdrop : c.poster, fit: BoxFit.cover),
              const DecoratedBox(
                decoration: BoxDecoration(
                  gradient: LinearGradient(begin: Alignment.topCenter, end: Alignment.bottomCenter,
                      colors: [Colors.transparent, Color(0xFF0B0B10)]),
                ),
              ),
            ]),
          ),
        ),
        SliverPadding(
          padding: const EdgeInsets.all(16),
          sliver: SliverList(
            delegate: SliverChildListDelegate([

import 'package:chewie/chewie.dart';
import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:video_player/video_player.dart';

class PlayerView extends StatefulWidget {
  final String url, title;
  const PlayerView({super.key, required this.url, required this.title});
  @override
  State<PlayerView> createState() => _PlayerViewState();
}

class _PlayerViewState extends State<PlayerView> {
  late final VideoPlayerController vc;
  ChewieController? cc;

  @override
  void initState() {
    super.initState();
    SystemChrome.setPreferredOrientations([DeviceOrientation.landscapeLeft, DeviceOrientation.landscapeRight]);
    vc = VideoPlayerController.networkUrl(Uri.parse(widget.url));
    vc.initialize().then((_) {
      if (!mounted) return;
      setState(() => cc = ChewieController(videoPlayerController: vc, autoPlay: true, aspectRatio: vc.value.aspectRatio));
    });
  }

  @override
  void dispose() {
    cc?.dispose();
    vc.dispose();
    SystemChrome.setPreferredOrientations(DeviceOrientation.values);
    super.dispose();
  }

  @override
  Widget build(BuildContext context) => Scaffold(
        backgroundColor: Colors.black,
        body: Stack(children: [
          Center(child: cc == null ? const CircularProgressIndicator() : Chewie(controller: cc!)),
          SafeArea(child: BackButton(color: Colors.white)),
        ]),
      );
}


==================== FILE: README.md ====================

# Cheetah BD

## 1. Backend (Vercel)
Push `api/`, `requirements.txt`, `vercel.json` to GitHub -> import in Vercel -> deploy.
Test: https://YOUR-APP.vercel.app/api/get-link?url=VIDEO_URL
Then set `kApiBase` in lib/services.dart.

## 2. Firebase
- Enable Authentication (Email/Password) + Firestore
- Android: put google-services.json in android/app/ (or run `flutterfire configure`)
- Collection `content`, each doc:
  title, type ("movie"|"series"), poster, backdrop, overview, year, rating (number),
  category ("Trending", "Bangla" ...), servers: [ {name: "Server 1", url: "https://..."} ]
- Rules (dev): allow read: if request.auth != null;

## 3. Run
flutter create .   (generates android/ios folders, keeps lib/)
flutter pub get && flutter run
flutter build apk --release
