final db = FirebaseFirestore.instance;

/// TMDB / OMDB access. Keys are stored in Firestore (config/api) and set
/// from the admin panel, so they never live in the source code.
class Api {
  static String tmdbKey = '';
  static String omdbKey = '';

  static Future<void> load() async {
    try {
      final d = await db.collection('config').doc('api').get();
      final m = d.data() ?? {};
      tmdbKey = '${m['tmdbKey'] ?? ''}'.trim();
      omdbKey = '${m['omdbKey'] ?? ''}'.trim();
    } catch (_) {}
  }

  static Future<Map<String, dynamic>?> tmdb(String path,
      [Map<String, String> q = const {}]) async {
    if (tmdbKey.isEmpty) return null;
    try {
      final uri =
          Uri.https('api.themoviedb.org', '/3/$path', {'api_key': tmdbKey, ...q});
      final r = await http.get(uri).timeout(const Duration(seconds: 15));
      if (r.statusCode == 200) {
        return Map<String, dynamic>.from(jsonDecode(r.body) as Map);
      }
    } catch (_) {}
    return null;
  }

  static Future<Map<String, dynamic>?> omdb(String imdbId) async {
    if (omdbKey.isEmpty || imdbId.isEmpty) return null;
    try {
      final uri = Uri.https('www.omdbapi.com', '/', {'apikey': omdbKey, 'i': imdbId});
      final r = await http.get(uri).timeout(const Duration(seconds: 15));
      if (r.statusCode == 200) {
        final j = Map<String, dynamic>.from(jsonDecode(r.body) as Map);
        if (j['Response'] == 'True') return j;
      }
    } catch (_) {}
    return null;
  }
}

class _Col {
  final String name;
  final ValueNotifier<List<Item>> notifier;
  final int Function(Item, Item) cmp;
  _Col(this.name, this.notifier, this.cmp);
}

/// Live copies of the Firestore collections the app needs.
class Store {
  static final content = ValueNotifier<List<Item>>([]);
  static final slides = ValueNotifier<List<Item>>([]);
  static final ads = ValueNotifier<List<Item>>([]);
  static final servers = ValueNotifier<List<Item>>([]);
  static final popups = ValueNotifier<List<Item>>([]);
  static final notifs = ValueNotifier<List<Item>>([]);

  /// false until the first batch of content arrives (drives the skeletons).
  static final ready = ValueNotifier<bool>(false);

  static final List<StreamSubscription> _subs = [];

  static int _ts(Item m) {
    final t = m['createdAt'];
    return t is Timestamp ? t.millisecondsSinceEpoch : 0;
  }

  static num _n(dynamic v) => v is num ? v : 0;

  static int _newest(Item a, Item b) => _ts(b).compareTo(_ts(a));

  static int _byOrder(Item a, Item b) => _n(a['order']).compareTo(_n(b['order']));

  static final List<_Col> _cols = [
    _Col('content', content, _newest),
    _Col('slides', slides, _byOrder),
    _Col('ads', ads, _newest),
    _Col('servers', servers, _byOrder),
    _Col('popups', popups, _newest),
    _Col('notifications', notifs, _newest),
  ];

  static List<Item> _read(QuerySnapshot<Map<String, dynamic>> s, int Function(Item, Item) cmp) {
    final l = s.docs.map((d) => <String, dynamic>{...d.data(), 'id': d.id}).toList();
    l.sort(cmp);
    return l;
  }

  static void start() {
    if (_subs.isNotEmpty) return;
    for (final c in _cols) {
      _subs.add(db.collection(c.name).snapshots().listen((s) {
        c.notifier.value = _read(s, c.cmp);
        if (c.name == 'content') ready.value = true;
      }, onError: (_) {
        if (c.name == 'content') ready.value = true;
      }));
    }
  }

  static void stop() {
    for (final s in _subs) {
      s.cancel();
    }
    _subs.clear();
    ready.value = false;
  }

  static Future<bool> _fetch(_Col c) async {
    try {
      final s = await db
          .collection(c.name)
          .get(const GetOptions(source: Source.server))
          .timeout(const Duration(seconds: 15));
      c.notifier.value = _read(s, c.cmp);
      if (c.name == 'content') ready.value = true;
      return true;
    } catch (_) {
      return false;
    }
  }

  /// Pull-to-refresh: re-reads every collection straight from the server and
  /// reloads the API keys. Returns false if the server could not be reached.
  static Future<bool> refresh() async {
    final r = await Future.wait<bool>([
      for (final c in _cols) _fetch(c),
      Api.load().then((_) => true),
    ]);
    return r.every((e) => e);
  }
}

Future<void> openLink(String url) async {
  if (url.trim().isEmpty) return;
  try {
    final u = Uri.tryParse(url.trim());
    if (u != null) await launchUrl(u, mode: LaunchMode.externalApplication);
  } catch (_) {}
}

void openItem(BuildContext c, Item it) => Navigator.push(
      c,
      PageRouteBuilder(
        transitionDuration: const Duration(milliseconds: 280),
        reverseTransitionDuration: const Duration(milliseconds: 220),
        pageBuilder: (_, __, ___) => DetailScreen(item: it),
        transitionsBuilder: (_, a, __, child) => FadeTransition(
          opacity: CurvedAnimation(parent: a, curve: Curves.easeOut),
          child: SlideTransition(
            position: Tween(begin: const Offset(0, 0.04), end: Offset.zero)
                .animate(CurvedAnimation(parent: a, curve: Curves.easeOutCubic)),
            child: child,
          ),
        ),
      ),
    );

/// Wrap any scrollable (with AlwaysScrollableScrollPhysics) to get
/// pull-to-refresh that re-reads everything from the server.
class Pull extends StatelessWidget {
  final Widget child;
  const Pull({super.key, required this.child});

  Future<void> _go(BuildContext c) async {
    HapticFeedback.lightImpact();
    final ok = await Store.refresh();
    if (!c.mounted) return;
    final m = ScaffoldMessenger.of(c);
    m.hideCurrentSnackBar();
    m.showSnackBar(SnackBar(
      behavior: SnackBarBehavior.floating,
      duration: const Duration(seconds: 2),
      backgroundColor: kCard2,
      shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(14)),
      content: Row(children: [
        Icon(ok ? Icons.check_circle_rounded : Icons.wifi_off_rounded,
            size: 18, color: ok ? Colors.greenAccent : Colors.orangeAccent),
        const SizedBox(width: 10),
        Text(ok ? 'Up to date' : 'Couldn\'t reach the server'),
      ]),
    ));
  }

  @override
  Widget build(BuildContext context) {
    return RefreshIndicator(
      color: kRed,
      backgroundColor: kCard2,
      displacement: 48,
      onRefresh: () => _go(context),
      child: child,
    );
  }
}

class Skeleton extends StatefulWidget {
  final double? width;
  final double height;
  final double radius;
  const Skeleton({super.key, this.width, required this.height, this.radius = 12});

  @override
  State<Skeleton> createState() => _SkeletonState();
}

class _SkeletonState extends State<Skeleton> with SingleTickerProviderStateMixin {
  late final AnimationController _c =
      AnimationController(vsync: this, duration: const Duration(milliseconds: 1300))..repeat();

  @override
  void dispose() {
    _c.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return AnimatedBuilder(
      animation: _c,
      builder: (_, __) {
        final t = _c.value;
        return Container(
          width: widget.width,
          height: widget.height,
          decoration: BoxDecoration(
            borderRadius: BorderRadius.circular(widget.radius),
            gradient: LinearGradient(
              begin: Alignment(-1.5 + 3 * t, 0),
              end: Alignment(-0.5 + 3 * t, 0),
              colors: const [kCard, kCard2, kCard],
            ),
          ),
        );
      },
    );
  }
}

    );
  }
}

class _AuthScreenState extends State<AuthScreen> with TickerProviderStateMixin {
  bool login = true;
  bool busy = false;
  bool hide = true;
  String? err;
  final name = TextEditingController();
  final email = TextEditingController();
  final pass = TextEditingController();

  late final AnimationController _bg =
      AnimationController(vsync: this, duration: const Duration(seconds: 14))..repeat();
  late final AnimationController _enter =
      AnimationController(vsync: this, duration: const Duration(milliseconds: 1100))..forward();
  late final AnimationController _shine =
      AnimationController(vsync: this, duration: const Duration(milliseconds: 2400))..repeat();

  @override
  void dispose() {
    _bg.dispose();
    _enter.dispose();
    _shine.dispose();
    name.dispose();
    email.dispose();
    pass.dispose();
    super.dispose();
  }

  String _msg(FirebaseAuthException e) {
    switch (e.code) {
      case 'invalid-email':
        return 'Please enter a valid email address.';
      case 'user-not-found':
      case 'wrong-password':
      case 'invalid-credential':
        return 'Incorrect email or password.';
      case 'email-already-in-use':
        return 'An account with this email already exists.';
      case 'weak-password':
        return 'Password must be at least 6 characters.';
      case 'network-request-failed':
        return 'No internet connection.';
      case 'too-many-requests':
        return 'Too many attempts. Try again later.';
      default:
        return e.message ?? 'Something went wrong.';
    }
  }

  Future<void> submit() async {
    FocusScope.of(context).unfocus();
    setState(() {
      busy = true;
      err = null;
    });
    try {
      final auth = FirebaseAuth.instance;
      if (login) {
        await auth.signInWithEmailAndPassword(email: email.text.trim(), password: pass.text);
      } else {
        if (name.text.trim().isEmpty) {
          throw FirebaseAuthException(code: 'custom', message: 'Please enter your name.');
        }
        final cr = await auth.createUserWithEmailAndPassword(
            email: email.text.trim(), password: pass.text);
        await cr.user?.updateDisplayName(name.text.trim());
        try {
          await db.collection('users').doc(cr.user!.uid).set({
            'name': name.text.trim(),
            'email': email.text.trim(),
            'createdAt': FieldValue.serverTimestamp(),
          });
        } catch (_) {}
      }
    } on FirebaseAuthException catch (e) {
      err = _msg(e);
    } catch (e) {
      err = 'Something went wrong. Try again.';
    }
    if (mounted) setState(() => busy = false);
  }

  /// Staggered fade + slide-up for each block of the page.
  Widget _stagger(int i, Widget child) {
    final a = CurvedAnimation(
      parent: _enter,
      curve: Interval((0.08 * i).clamp(0.0, 0.7), (0.45 + 0.08 * i).clamp(0.2, 1.0), curve: Curves.easeOutCubic),
    );
    return AnimatedBuilder(
      animation: a,
      builder: (_, c) => Opacity(
        opacity: a.value,
        child: Transform.translate(offset: Offset(0, 28 * (1 - a.value)), child: c),
      ),
      child: child,
    );
  }

  Widget _blob(double t, double phase, Color c, double size, Alignment base) {
    final a = t * 2 * math.pi + phase;
    return Align(
      alignment: Alignment(base.x + 0.35 * math.sin(a), base.y + 0.25 * math.cos(a * 0.9)),
      child: Container(
        width: size,
        height: size,
        decoration: BoxDecoration(
          shape: BoxShape.circle,
          gradient: RadialGradient(colors: [c.withOpacity(0.55), c.withOpacity(0)]),
        ),
      ),
    );
  }

class _ShellState extends State<Shell> {
  int tab = 0;
  bool _popupDone = false;

  @override
  void initState() {
    super.initState();
    Store.popups.addListener(_checkPopup);
    WidgetsBinding.instance.addPostFrameCallback((_) => _checkPopup());
  }

  @override
  void dispose() {
    Store.popups.removeListener(_checkPopup);
    super.dispose();
  }

  Future<void> _checkPopup() async {
    if (_popupDone || !mounted) return;
    final list = Store.popups.value.where((p) => p['active'] == true).toList();
    if (list.isEmpty) return;
    _popupDone = true;
    final p = list.first;
    final always = p['showAlways'] == true;
    final sp = await SharedPreferences.getInstance();
    final seen = sp.getStringList('seen_popups') ?? <String>[];
    if (!always && seen.contains('${p['id']}')) return;
    if (!mounted) return;
    await showDialog(
      context: context,
      builder: (ctx) => Dialog(
        backgroundColor: kCard,
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(20)),
        clipBehavior: Clip.antiAlias,
        child: SingleChildScrollView(
          child: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              if ('${p['image'] ?? ''}'.isNotEmpty)
                AspectRatio(aspectRatio: 16 / 9, child: Img(p['image']?.toString())),
              Padding(
                padding: const EdgeInsets.all(18),
                child: Column(
                  children: [
                    if ('${p['title'] ?? ''}'.isNotEmpty)
                      Text('${p['title']}',
                          textAlign: TextAlign.center,
                          style: const TextStyle(fontSize: 18, fontWeight: FontWeight.w800)),
                    if ('${p['message'] ?? ''}'.isNotEmpty) ...[
                      const SizedBox(height: 8),
                      Text('${p['message']}',
                          textAlign: TextAlign.center, style: const TextStyle(color: Colors.white70)),
                    ],
                    const SizedBox(height: 16),
                    Row(
                      children: [
                        Expanded(
                          child: OutlinedButton(
                            onPressed: () => Navigator.pop(ctx),
                            child: const Text('Close'),
                          ),
                        ),
                        if ('${p['link'] ?? ''}'.isNotEmpty) ...[
                          const SizedBox(width: 10),
                          Expanded(
                            child: FilledButton(
                              style: FilledButton.styleFrom(backgroundColor: kRed),
                              onPressed: () {
                                Navigator.pop(ctx);
                                openLink('${p['link']}');
                              },
                              child: Text('${p['buttonText'] ?? ''}'.isEmpty ? 'Open' : '${p['buttonText']}'),
                            ),
                          ),
                        ],
                      ],
                    ),
                  ],
                ),
              ),
            ],
          ),
        ),
      ),
    );
    if (!always) {
      seen.add('${p['id']}');
      await sp.setStringList('seen_popups', seen);
    }
  }

class _HeroSliderState extends State<HeroSlider> {
  static const _vf = 0.88;
  final _pc = PageController(viewportFraction: _vf);
  Timer? _t;
  int _i = 0;
  DateTime _touched = DateTime.fromMillisecondsSinceEpoch(0);

  @override
  void initState() {
    super.initState();
    _t = Timer.periodic(const Duration(seconds: 5), (_) {
      if (!_pc.hasClients) return;
      if (DateTime.now().difference(_touched) < const Duration(seconds: 7)) return;
      final n = _items().length;
      if (n < 2) return;
      _pc.animateToPage((_i + 1) % n,
          duration: const Duration(milliseconds: 550), curve: Curves.easeInOutCubic);
    });
  }

  @override
  void dispose() {
    _t?.cancel();
    _pc.dispose();
    super.dispose();
  }

  Item? _content(String id) =>
      id.isEmpty ? null : Store.content.value.where((e) => e['id'] == id).firstOrNull;

  String _meta(Item? it) {
    if (it == null) return '';
    final p = <String>[];
    final y = '${it['year'] ?? ''}';
    if (y.isNotEmpty) p.add(y);
    p.add(it['type'] == 'tv' ? 'Series' : (it['type'] == 'movie' ? 'Movie' : 'Featured'));
    final r = it['rating'] is num ? (it['rating'] as num).toDouble() : 0.0;
    if (r > 0) p.add('★ ${r.toStringAsFixed(1)}');
    return p.join('  ·  ');
  }

  List<_Slide> _items() {
    final s = Store.slides.value
        .where((e) => e['active'] != false && '${e['image'] ?? ''}'.isNotEmpty)
        .toList();
    if (s.isNotEmpty) {
      return s.map((e) {
        final id = '${e['contentId'] ?? ''}';
        return _Slide('${e['image']}', '${e['title'] ?? ''}', id, '${e['link'] ?? ''}', _meta(_content(id)));
      }).toList();
    }
    return Store.content.value
        .where((e) => '${e['backdrop'] ?? ''}'.isNotEmpty)
        .take(5)
        .map((e) => _Slide('${e['backdrop']}', '${e['title'] ?? ''}', '${e['id']}', '', _meta(e)))
        .toList();
  }

  void _tap(_Slide s) {
    final it = _content(s.contentId);
    if (it != null) {
      openItem(context, it);
      return;
    }
    openLink(s.link);
  }

  @override
  Widget build(BuildContext context) {
    return AnimatedBuilder(
      animation: Listenable.merge([Store.slides, Store.content]),
      builder: (_, __) {
        final items = _items();
        if (items.isEmpty) return const SizedBox.shrink();
        return LayoutBuilder(
          builder: (_, box) {
            final h = box.maxWidth * _vf * 10 / 16;
            return Column(
              children: [
                SizedBox(
                  height: h,
                  child: Listener(
                    onPointerDown: (_) => _touched = DateTime.now(),
                    child: PageView.builder(
                      controller: _pc,
                      physics: const PageScrollPhysics(parent: BouncingScrollPhysics()),
                      itemCount: items.length,
                      onPageChanged: (i) => setState(() => _i = i),
                      itemBuilder: (_, i) => AnimatedBuilder(
                        animation: _pc,
                        builder: (_, child) {
                          var page = _i.toDouble();
                          if (_pc.hasClients && _pc.position.haveDimensions) page = _pc.page ?? page;
                          final d = (page - i).abs().clamp(0.0, 1.0);
                          return Transform.scale(
                            scale: 1 - d * 0.07,
                            child: Opacity(opacity: 1 - d * 0.35, child: child),
                          );
                        },
                        child: _slide(items[i]),
                      ),
                    ),
                  ),
                ),
                const SizedBox(height: 12),
                Row(
                  mainAxisAlignment: MainAxisAlignment.center,
                  children: [
                    for (var i = 0; i < items.length; i++)
                      AnimatedContainer(
                        duration: const Duration(milliseconds: 280),
                        curve: Curves.easeOutCubic,
                        margin: const EdgeInsets.symmetric(horizontal: 3),
                        width: i == _i ? 22 : 6,
                        height: 6,
                        decoration: BoxDecoration(
                          gradient: i == _i ? kRedGradient : null,
                          color: i == _i ? null : Colors.white24,
                          borderRadius: BorderRadius.circular(3),
                        ),
                      ),
                  ],
                ),
              ],
            );
          },
        );
      },
    );
  }

  Widget _slide(_Slide s) {
    return Padding(
      padding: const EdgeInsets.symmetric(horizontal: 5),
      child: GestureDetector(
        onTap: () => _tap(s),
        child: Container(
          decoration: BoxDecoration(
            borderRadius: BorderRadius.circular(22),
            border: Border.all(color: Colors.white12),

          ),

                const DecoratedBox(

                    gradient: LinearGradient(
                      begin: Alignment.topCenter,
                      end: Alignment.bottomCenter,
                      colors: [Colors.transparent, Color(0xF2000000)],
                      stops: [0.35, 1],
                    ),
                  ),
                ),
                Positioned(
                  left: 16,
                  right: 16,
                  bottom: 14,
                  child: Row(
                    crossAxisAlignment: CrossAxisAlignment.end,
                    children: [
                      Expanded(
                        child: Column(
                          crossAxisAlignment: CrossAxisAlignment.start,
                          mainAxisSize: MainAxisSize.min,
                          children: [
                            if (s.title.isNotEmpty)
                              Text(
                                s.title,
                                maxLines: 2,
                                overflow: TextOverflow.ellipsis,
                                style: const TextStyle(fontSize: 19, fontWeight: FontWeight.w800, height: 1.15),
                              ),
                            if (s.meta.isNotEmpty) ...[

class _SearchScreenState extends State<SearchScreen> {
  final _ctl = TextEditingController();
  Timer? _debounce;
  String q = '';
  List<String> recents = [];

  @override
  void initState() {
    super.initState();
    SharedPreferences.getInstance().then((sp) {
      if (mounted) setState(() => recents = sp.getStringList('recent_searches') ?? []);
    });
  }

  @override
  void dispose() {
    _debounce?.cancel();
    _ctl.dispose();
    super.dispose();
  }

  void _onChanged(String v) {
    _debounce?.cancel();
    _debounce = Timer(const Duration(milliseconds: 220), () {
      if (mounted) setState(() => q = v.trim().toLowerCase());
    });
    setState(() {}); // refresh the clear button immediately
  }

  Future<void> _saveRecent(String v) async {
    final t = v.trim();
    if (t.isEmpty) return;
    final l = [t, ...recents.where((e) => e.toLowerCase() != t.toLowerCase())].take(8).toList();
    setState(() => recents = l);
    final sp = await SharedPreferences.getInstance();
    await sp.setStringList('recent_searches', l);
  }

  Future<void> _clearRecents() async {
    setState(() => recents = []);
    final sp = await SharedPreferences.getInstance();
    await sp.remove('recent_searches');
  }

  void _use(String s) {
    _ctl.text = s;
    _ctl.selection = TextSelection.collapsed(offset: s.length);
    setState(() => q = s.toLowerCase());
    FocusScope.of(context).unfocus();
  }

  bool _match(Item e) {
    if ('${e['title'] ?? ''}'.toLowerCase().contains(q)) return true;
    if ('${e['year'] ?? ''}' == q) return true;
    final g = e['genres'];
    return g is List && g.any((x) => '$x'.toLowerCase().contains(q));
  }

  @override
  Widget build(BuildContext context) {
    return SafeArea(
      bottom: false,
      child: Column(
        children: [
          Padding(
            padding: const EdgeInsets.fromLTRB(16, 14, 16, 8),
            child: TextField(
              controller: _ctl,
              onChanged: _onChanged,
              textInputAction: TextInputAction.search,
              onSubmitted: (v) {
                _debounce?.cancel();
                setState(() => q = v.trim().toLowerCase());
                _saveRecent(v);
              },
              decoration: InputDecoration(
                hintText: 'Search movies, series, genres…',
                hintStyle: const TextStyle(color: Colors.white38),
                prefixIcon: const Icon(Icons.search_rounded),
                suffixIcon: _ctl.text.isEmpty
                    ? null
                    : IconButton(
                        icon: const Icon(Icons.close_rounded, size: 20),
                        onPressed: () {
                          _ctl.clear();
                          _debounce?.cancel();
                          setState(() => q = '');
                        },
                      ),
                filled: true,
                fillColor: kCard,
                contentPadding: const EdgeInsets.symmetric(vertical: 14),
                border: OutlineInputBorder(borderRadius: BorderRadius.circular(16), borderSide: BorderSide.none),
                focusedBorder: OutlineInputBorder(
                  borderRadius: BorderRadius.circular(16),
                  borderSide: const BorderSide(color: kRed, width: 1.2),
                ),
              ),
            ),
          ),
          Expanded(
            child: ValueListenableBuilder<List<Item>>(
              valueListenable: Store.content,
              builder: (_, all, __) {
                if (q.isEmpty) return _idle(all);
                final res = all.where(_match).toList();
                if (res.isEmpty) {
                  return emptyScrollable(EmptyState(
                    icon: Icons.search_off_rounded,
                    title: 'No results',
                    subtitle: 'Nothing matched “${_ctl.text.trim()}”.\nTry a different title or genre.',
                  ));
                }
                return Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    Padding(
                      padding: const EdgeInsets.fromLTRB(18, 4, 16, 0),
                      child: Text('${res.length} result${res.length == 1 ? '' : 's'}',
                          style: const TextStyle(color: Colors.white54, fontSize: 13)),
                    ),
                    Expanded(child: Pull(child: posterGrid(res))),
                  ],
                );
              },
            ),
          ),
        ],

  }

  Widget _idle(List<Item> all) {
    final top = all.where((e) => e['rating'] is num && (e['rating'] as num) > 0).toList()

                  Padding(

                    child: Wrap(

  bool playing = false;

  bool get isTv => it['type'] == 'tv';
  String get tmdbId => '${it['tmdbId'] ?? ''}';

  @override
  void initState() {
    super.initState();
    // Resume from the last watched episode.
    final h = UserData.history.value['${it['id']}'];
    if (h != null && isTv) {
      season = (h['season'] as num?)?.toInt() ?? 1;
      episode = (h['episode'] as num?)?.toInt() ?? 1;
    }
    _loadExtra();
  }

  Future<void> _loadExtra() async {
    if (tmdbId.isEmpty) return;
    String imdb = '';
    if (isTv) {
      final d = await Api.tmdb('tv/$tmdbId', {'append_to_response': 'external_ids'});
      if (d == null) return;
      imdb = '${d['external_ids']?['imdb_id'] ?? ''}';
      final n = (d['number_of_seasons'] as num?)?.toInt() ?? 1;
      if (mounted) setState(() => seasonCount = n < 1 ? 1 : n);
      _loadEpisodes();
    } else {
      final d = await Api.tmdb('movie/$tmdbId');
      imdb = '${d?['imdb_id'] ?? ''}';
    }
    final o = await Api.omdb(imdb);
    if (mounted && o != null) setState(() => omdb = o);
  }

  Future<void> _loadEpisodes() async {
    final d = await Api.tmdb('tv/$tmdbId/season/$season');
    final eps = d?['episodes'];
    if (eps is List && mounted) {
      setState(() {
        epCount = eps.isEmpty ? 1 : eps.length;
        epNames = [for (final e in eps) '${e is Map ? (e['name'] ?? '') : ''}'];
        if (episode > epCount) episode = 1;
      });
    }
  }

  String _fill(String tpl) => tpl
      .replaceAll('{tmdb_id}', tmdbId)
      .replaceAll('{season}', '$season')
      .replaceAll('{episode}', '$episode');

    final l = <Map<String, String>>[];
    if (tmdbId.isNotEmpty) {
      for (final s in Store.servers.value.where((s) => s['active'] != false)) {
        final tpl = '${(isTv ? s['tvUrl'] : s['movieUrl']) ?? ''}';
        if (tpl.isEmpty) continue;
        l.add({'name': '${s['name'] ?? 'Server'}', 'url': _fill(tpl)});
      }
    }
    final own = it['servers'];
    if (own is List) {
      for (final m in own) {
        if (m is Map && '${m['url'] ?? ''}'.isNotEmpty) {
          l.add({'name': '${m['name'] ?? 'Server'}', 'url': _fill('${m['url']}')});
        }
      }
    }
    return l;

  /// Titles that share genres with this one.
  List<Item> get related {
    final g = it['genres'];
    final mine = g is List ? g.map((e) => '$e').toSet() : <String>{};
    if (mine.isEmpty) return [];
    int score(Item e) {
      final x = e['genres'];
      return x is List ? x.where((a) => mine.contains('$a')).length : 0;
    }

    final l = Store.content.value.where((e) => e['id'] != it['id'] && score(e) > 0).toList();
    l.sort((a, b) => score(b).compareTo(score(a)));
    return l.take(12).toList();
  }

class PlayerView extends StatefulWidget {

  const PlayerView({super.key, required this.url});

  @override
  State<PlayerView> createState() => _PlayerViewState();

class _PlayerViewState extends State<PlayerView> {
  late final WebViewController c;
  bool loading = true;
  bool loaded = false;

  String _root(String h) {
    final p = h.split('.');
    return p.length >= 2 ? p.sublist(p.length - 2).join('.') : h;
  }

  @override
  void initState() {
    super.initState();
    final base = _root(Uri.tryParse(widget.url)?.host ?? '');
    c = WebViewController()
      ..setJavaScriptMode(JavaScriptMode.unrestricted)
      ..setBackgroundColor(Colors.black)
      ..setNavigationDelegate(NavigationDelegate(
        onPageFinished: (_) {
          loaded = true;
          if (mounted) setState(() => loading = false);
        },
        onNavigationRequest: (r) {
          if (!loaded || !r.isMainFrame) return NavigationDecision.navigate;
          final u = Uri.tryParse(r.url);
          if (u == null) return NavigationDecision.prevent;
          if (u.scheme == 'about' || u.scheme == 'data' || _root(u.host) == base) {
            return NavigationDecision.navigate;
          }
          return NavigationDecision.prevent;
        },
      ));
    final lower = widget.url.toLowerCase().split('?').first;
    if (lower.endsWith('.mp4') || lower.endsWith('.webm')) {
      c.loadHtmlString(
          '<html><body style="margin:0;background:#000"><video src="${widget.url}" controls autoplay playsinline style="width:100%;height:100%"></video></body></html>');
    } else {
      c.loadRequest(Uri.parse(widget.url));
    }
  }

  @override
  Widget build(BuildContext context) {
    return Stack(
      fit: StackFit.expand,
      children: [
        WebViewWidget(controller: c),
        if (loading) const Center(child: CircularProgressIndicator()),
      ],
    );
  }
}

import 'package:cloud_firestore/cloud_firestore.dart';

/// Per-user data: likes, favorites, watchlist, watch history.
/// Stored under users/{uid}/{likes|favorites|watchlist|history}/{contentId}.
class UserData {
  static final likes = ValueNotifier<Map<String, Item>>({});
  static final favorites = ValueNotifier<Map<String, Item>>({});
  static final watchlist = ValueNotifier<Map<String, Item>>({});
  static final history = ValueNotifier<Map<String, Item>>({});
  static final profile = ValueNotifier<Item>({});

  static final List<StreamSubscription> _subs = [];
  static String? _uid;

  static DocumentReference<Map<String, dynamic>>? get _me =>
      _uid == null ? null : db.collection('users').doc(_uid);

  static void start() {
    final u = FirebaseAuth.instance.currentUser;
    if (u == null || _uid == u.uid) return;
    stop();
    _uid = u.uid;
    final me = _me!;
    _listen('likes', likes);
    _listen('favorites', favorites);
    _listen('watchlist', watchlist);
    _listen('history', history);
    _subs.add(me.snapshots().listen((d) => profile.value = d.data() ?? {}, onError: (_) {}));
  }

  static void _listen(String name, ValueNotifier<Map<String, Item>> n) {
    _subs.add(_me!.collection(name).snapshots().listen((s) {
      n.value = {for (final d in s.docs) d.id: <String, dynamic>{...d.data(), 'id': d.id}};
    }, onError: (_) {}));
  }

  static void stop() {
    for (final s in _subs) {
      s.cancel();
    }
    _subs.clear();
    _uid = null;
    likes.value = {};
    favorites.value = {};
    watchlist.value = {};
    history.value = {};
    profile.value = {};
  }

  /// Small snapshot of a title, enough to draw a card without the content list.
  static Map<String, dynamic> _snap(Item it) => {
        'title': it['title'],
        'poster': it['poster'],
        'backdrop': it['backdrop'],
        'type': it['type'],
        'year': it['year'],
        'rating': it['rating'],
      };

  static Future<void> _toggle(String col, Item it, bool on) async {
    final me = _me;
    if (me == null) return;
    final ref = me.collection(col).doc('${it['id']}');
    if (on) {
      await ref.delete();
    } else {
      await ref.set({..._snap(it), 'at': FieldValue.serverTimestamp()});
    }
  }

  static bool isLiked(Item it) => likes.value.containsKey('${it['id']}');
  static bool isFav(Item it) => favorites.value.containsKey('${it['id']}');
  static bool inList(Item it) => watchlist.value.containsKey('${it['id']}');

  static Future<void> toggleLike(Item it) => _toggle('likes', it, isLiked(it));
  static Future<void> toggleFav(Item it) => _toggle('favorites', it, isFav(it));
  static Future<void> toggleList(Item it) => _toggle('watchlist', it, inList(it));

  /// Called when playback starts / the episode changes.
  static Future<void> addHistory(Item it, {int season = 1, int episode = 1}) async {
    final me = _me;
    if (me == null) return;
    try {
      await me.collection('history').doc('${it['id']}').set({
        ..._snap(it),
        'season': season,
        'episode': episode,
        'at': FieldValue.serverTimestamp(),
      });
    } catch (_) {}
  }

  static Future<void> removeHistory(String id) async => _me?.collection('history').doc(id).delete();

  static Future<void> clearHistory() async {
    final me = _me;
    if (me == null) return;
    final s = await me.collection('history').get();
    final b = db.batch();
    for (final d in s.docs) {
      b.delete(d.reference);
    }
    await b.commit();
  }

  static Future<void> saveProfile(Map<String, dynamic> m) async =>
      _me?.set(m, SetOptions(merge: true));

  static int _ms(Item m) {
    final t = m['at'];
    return t is Timestamp ? t.millisecondsSinceEpoch : DateTime.now().millisecondsSinceEpoch;
  }

  /// Newest first.
  static List<Item> sorted(Map<String, Item> m) {
    final l = m.values.toList();
    l.sort((a, b) => _ms(b).compareTo(_ms(a)));
    return l;
  }

  /// Full content item if it still exists, else the saved snapshot.
  static Item resolve(Item saved) {
    final full = Store.content.value.where((e) => e['id'] == saved['id']).firstOrNull;
    return full ?? saved;
  }
}
