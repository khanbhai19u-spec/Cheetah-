import 'dart:async';

import 'package:http/http.dart' as http;

typedef Item = Map<String, dynamic>;

final db = FirebaseFirestore.instance;

class Api {
  static String tmdbKey = '';
  static String omdbKey = '';

  static Future<void> load() async {
    try {
      final d = await db.collection('config').doc('api').get();
      final m = d.data() ?? {};
      tmdbKey = '${m['tmdbKey'] ?? ''}'.trim();
      omdbKey = '${m['omdbKey'] ?? ''}'.trim();
    } catch (_) {}
  }

  static Future<Map<String, dynamic>?> tmdb(String path,
      [Map<String, String> q = const {}]) async {
    if (tmdbKey.isEmpty) return null;
    try {
      final uri =
          Uri.https('api.themoviedb.org', '/3/$path', {'api_key': tmdbKey, ...q});
      final r = await http.get(uri).timeout(const Duration(seconds: 15));
      if (r.statusCode == 200) {
        return Map<String, dynamic>.from(jsonDecode(r.body) as Map);
      }
    } catch (_) {}
    return null;
  }

  static Future<Map<String, dynamic>?> omdb(String imdbId) async {
    if (omdbKey.isEmpty || imdbId.isEmpty) return null;
    try {
      final uri = Uri.https('www.omdbapi.com', '/', {'apikey': omdbKey, 'i': imdbId});
      final r = await http.get(uri).timeout(const Duration(seconds: 15));
      if (r.statusCode == 200) {
        final j = Map<String, dynamic>.from(jsonDecode(r.body) as Map);
        if (j['Response'] == 'True') return j;
      }
    } catch (_) {}
    return null;
  }
}

class _Col {
  final String name;
  final ValueNotifier<List<Item>> notifier;
  final int Function(Item, Item) cmp;
  _Col(this.name, this.notifier, this.cmp);
}

class Store {
  static final content = ValueNotifier<List<Item>>([]);
  static final slides = ValueNotifier<List<Item>>([]);
  static final ads = ValueNotifier<List<Item>>([]);
  static final servers = ValueNotifier<List<Item>>([]);
  static final popups = ValueNotifier<List<Item>>([]);
  static final notifs = ValueNotifier<List<Item>>([]);

  static final ready = ValueNotifier<bool>(false);

  static final List<StreamSubscription> _subs = [];

  static int _ts(Item m) {
    final t = m['createdAt'];
    return t is Timestamp ? t.millisecondsSinceEpoch : 0;
  }

  static num _n(dynamic v) => v is num ? v : 0;

  static int _newest(Item a, Item b) => _ts(b).compareTo(_ts(a));

  static int _byOrder(Item a, Item b) => _n(a['order']).compareTo(_n(b['order']));

  static final List<_Col> _cols = [
    _Col('content', content, _newest),
    _Col('slides', slides, _byOrder),
    _Col('ads', ads, _newest),
    _Col('servers', servers, _byOrder),
    _Col('popups', popups, _newest),
    _Col('notifications', notifs, _newest),
  ];

  static List<Item> _read(QuerySnapshot<Map<String, dynamic>> s, int Function(Item, Item) cmp) {
    final l = s.docs.map((d) => <String, dynamic>{...d.data(), 'id': d.id}).toList();
    l.sort(cmp);
    return l;
  }

  static void start() {
    if (_subs.isNotEmpty) return;
    for (final c in _cols) {
      _subs.add(db.collection(c.name).snapshots().listen((s) {
        c.notifier.value = _read(s, c.cmp);
        if (c.name == 'content') ready.value = true;
      }, onError: (_) {
        if (c.name == 'content') ready.value = true;
      }));
    }
  }

  static void stop() {
    for (final s in _subs) {
      s.cancel();
    }
    _subs.clear();
    ready.value = false;
  }

  static Future<bool> _fetch(_Col c) async {
    try {
      final s = await db
          .collection(c.name)
          .get(const GetOptions(source: Source.server))
          .timeout(const Duration(seconds: 15));
      c.notifier.value = _read(s, c.cmp);
      if (c.name == 'content') ready.value = true;
      return true;
    } catch (_) {
      return false;
    }
  }

  static Future<bool> refresh() async {
    final r = await Future.wait<bool>([
      for (final c in _cols) _fetch(c),
      Api.load().then((_) => true),
    ]);
    return r.every((e) => e);
  }
}

Future<void> openLink(String url) async {
  if (url.trim().isEmpty) return;
  try {
    final u = Uri.tryParse(url.trim());
    if (u != null) await launchUrl(u, mode: LaunchMode.externalApplication);
  } catch (_) {}
}

void openItem(BuildContext c, Item it) => Navigator.push(
      c,
      PageRouteBuilder(
        transitionDuration: const Duration(milliseconds: 280),
        reverseTransitionDuration: const Duration(milliseconds: 220),
        pageBuilder: (_, __, ___) => DetailScreen(item: it),
        transitionsBuilder: (_, a, __, child) => FadeTransition(
          opacity: CurvedAnimation(parent: a, curve: Curves.easeOut),
          child: SlideTransition(
            position: Tween(begin: const Offset(0, 0.04), end: Offset.zero)
                .animate(CurvedAnimation(parent: a, curve: Curves.easeOutCubic)),
            child: child,
          ),
        ),
      ),
    );

class Pull extends StatelessWidget {
  final Widget child;
  const Pull({super.key, required this.child});

  Future<void> _go(BuildContext c) async {
    HapticFeedback.lightImpact();
    final ok = await Store.refresh();
    if (!c.mounted) return;
    final m = ScaffoldMessenger.of(c);
    m.hideCurrentSnackBar();
    m.showSnackBar(SnackBar(
      behavior: SnackBarBehavior.floating,
      duration: const Duration(seconds: 2),
      backgroundColor: kCard2,
      shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(14)),
      content: Row(children: [
        Icon(ok ? Icons.check_circle_rounded : Icons.wifi_off_rounded,
            size: 18, color: ok ? Colors.greenAccent : Colors.orangeAccent),
        const SizedBox(width: 10),
        Text(ok ? 'Up to date' : 'Couldn\'t reach the server'),
      ]),
    ));
  }

  @override
  Widget build(BuildContext context) {
    return RefreshIndicator(
      color: kRed,
      backgroundColor: kCard2,
      displacement: 48,
      onRefresh: () => _go(context),
      child: child,
    );
  }
}

class _SkeletonState extends State<Skeleton> with SingleTickerProviderStateMixin {
  late final AnimationController _c =
      AnimationController(vsync: this, duration: const Duration(milliseconds: 1300))..repeat();

  @override
  void dispose() {
    _c.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return AnimatedBuilder(
      animation: _c,
      builder: (_, __) {
        final t = _c.value;
        return Container(
          width: widget.width,
          height: widget.height,
          decoration: BoxDecoration(
            borderRadius: BorderRadius.circular(widget.radius),
            gradient: LinearGradient(
              begin: Alignment(-1.5 + 3 * t, 0),
              end: Alignment(-0.5 + 3 * t, 0),
              colors: const [kCard, kCard2, kCard],
            ),
          ),
        );
      },
    );
  }
}

class DetailScreen extends StatefulWidget {
  final Item item;
  const DetailScreen({super.key, required this.item});

  @override
  State<DetailScreen> createState() => _DetailScreenState();
}

class _DetailScreenState extends State<DetailScreen> {
  Item get it => widget.item;
  Map<String, dynamic>? omdb;
  int season = 1, episode = 1, seasonCount = 1, epCount = 1, serverIdx = 0;
  List<String> epNames = [];
  bool playing = false;

  bool get isTv => it['type'] == 'tv';
  String get tmdbId => '${it['tmdbId'] ?? ''}';

  @override
  void initState() {
    super.initState();
    final h = UserData.history.value['${it['id']}'];
    if (h != null && isTv) {
      season = (h['season'] as num?)?.toInt() ?? 1;
      episode = (h['episode'] as num?)?.toInt() ?? 1;
    }
    _loadExtra();
  }

  Future<void> _loadExtra() async {
    if (tmdbId.isEmpty) return;
    String imdb = '';
    if (isTv) {
      final d = await Api.tmdb('tv/$tmdbId', {'append_to_response': 'external_ids'});
      if (d == null) return;
      imdb = '${d['external_ids']?['imdb_id'] ?? ''}';
      final n = (d['number_of_seasons'] as num?)?.toInt() ?? 1;
      if (mounted) setState(() => seasonCount = n < 1 ? 1 : n);
      _loadEpisodes();
    } else {
      final d = await Api.tmdb('movie/$tmdbId');
      imdb = '${d?['imdb_id'] ?? ''}';
    }
    final o = await Api.omdb(imdb);
    if (mounted && o != null) setState(() => omdb = o);
  }

  Future<void> _loadEpisodes() async {
    final d = await Api.tmdb('tv/$tmdbId/season/$season');
    final eps = d?['episodes'];
    if (eps is List && mounted) {
      setState(() {
        epCount = eps.isEmpty ? 1 : eps.length;
        epNames = [for (final e in eps) '${e is Map ? (e['name'] ?? '') : ''}'];
        if (episode > epCount) episode = 1;
      });
    }
  }

  String _fill(String tpl) => tpl
      .replaceAll('{tmdb_id}', tmdbId)
      .replaceAll('{season}', '$season')
      .replaceAll('{episode}', '$episode');

  List<Map<String, String>> get servers {
    final l = <Map<String, String>>[];
    if (tmdbId.isNotEmpty) {
      for (final s in Store.servers.value.where((s) => s['active'] != false)) {
        final tpl = '${(isTv ? s['tvUrl'] : s['movieUrl']) ?? ''}';
        if (tpl.isEmpty) continue;
        l.add({'name': '${s['name'] ?? 'Server'}', 'url': _fill(tpl)});
      }
    }
    final own = it['servers'];
    if (own is List) {
      for (final m in own) {
        if (m is Map && '${m['url'] ?? ''}'.isNotEmpty) {
          l.add({'name': '${m['name'] ?? 'Server'}', 'url': _fill('${m['url']}')});
        }
      }
    }
    return l;
  }

  void _showDownloadOptions() {
    final sv = servers;
    if (sv.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('No download source available.')),
      );
      return;
    }

    showModalBottomSheet(
      context: context,
      backgroundColor: kCard,
      shape: const RoundedRectangleBorder(
        borderRadius: BorderRadius.vertical(top: Radius.circular(20)),
      ),
      builder: (context) {
        return Container(
          padding: const EdgeInsets.all(20),
          child: Column(
            mainAxisSize: MainAxisSize.min,
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              const Text(
                'Select Quality & Server to Download',
                style: TextStyle(fontSize: 18, fontWeight: FontWeight.w800),
              ),
              const SizedBox(height: 14),
              Flexible(
                child: ListView.builder(
                  shrinkWrap: true,
                  itemCount: sv.length,
                  itemBuilder: (context, index) {
                    final server = sv[index];
                    return ListTile(
                      leading: const Icon(Icons.download_rounded, color: Colors.greenAccent),
                      title: Text(server['name'] ?? 'Quality / Server ${index + 1}'),
                      subtitle: const Text('Tap to download inside app', style: TextStyle(fontSize: 11, color: Colors.white54)),
                      trailing: const Icon(Icons.arrow_forward_ios_rounded, size: 14),
                      onTap: () async {
                        Navigator.pop(context);
                        final uri = Uri.parse(server['url']!);
                        if (await canLaunchUrl(uri)) {
                          await launchUrl(uri, mode: LaunchMode.inAppWebView);
                        }
                      },
                    );
                  },
                ),
              ),
            ],
          ),
        );
      },
    );
  }

  Widget _pill(String t, {IconData? icon, Color? color}) => Container(
        padding: const EdgeInsets.symmetric(horizontal: 10, vertical: 5),
        decoration: BoxDecoration(
          color: kCard,
          borderRadius: BorderRadius.circular(20),
          border: Border.all(color: Colors.white10),
        ),
        child: Row(
          mainAxisSize: MainAxisSize.min,
          children: [
            if (icon != null) ...[Icon(icon, size: 14, color: color), const SizedBox(width: 4)],
            Text(t, style: const TextStyle(fontSize: 12, color: Colors.white70)),
          ],
        ),
      );

  Future<void> _run(Future<void> f) async {
    try {
      await f.timeout(const Duration(seconds: 8));
    } catch (e) {
      if (!mounted) return;
      final t = '$e';
      ScaffoldMessenger.of(context).showSnackBar(SnackBar(
        content: Text('Could not save: ${t.length > 90 ? t.substring(0, 90) : t}'),
      ));
    }
  }

  Widget _action(IconData on, IconData off, String label, bool active, Color activeColor, VoidCallback tap) {
    return Expanded(
      child: GestureDetector(
        behavior: HitTestBehavior.opaque,
        onTap: () {
          HapticFeedback.lightImpact();
          tap();
        },
        child: Column(
          children: [
            AnimatedContainer(
              duration: const Duration(milliseconds: 220),
              width: 52,
              height: 52,
              decoration: BoxDecoration(
                color: active ? activeColor.withOpacity(0.18) : kCard,
                shape: BoxShape.circle,
                border: Border.all(color: active ? activeColor : Colors.white10, width: 1.4),
              ),
              child: AnimatedSwitcher(
                duration: const Duration(milliseconds: 200),
                child: Icon(active ? on : off, key: ValueKey(active), color: active ? activeColor : Colors.white70, size: 24),
              ),
            ),
            const SizedBox(height: 6),
            Text(label, style: TextStyle(fontSize: 12, color: active ? activeColor : Colors.white60, fontWeight: FontWeight.w600)),
          ],
        ),
      ),
    );
  }

  Widget _actions() {
    return AnimatedBuilder(
      animation: Listenable.merge([UserData.likes, UserData.favorites, UserData.watchlist]),

        children: [
          _action(Icons.thumb_up_rounded, Icons.thumb_up_outlined, 'Like', UserData.isLiked(it), const Color(0xFF29B6F6), () => _run(UserData.toggleLike(it))),
          _action(Icons.favorite_rounded, Icons.favorite_border_rounded, 'Favorite', UserData.isFav(it), kRedLight, () => _run(UserData.toggleFav(it))),
          _action(Icons.bookmark_rounded, Icons.bookmark_border_rounded, 'Watchlist', UserData.inList(it), const Color(0xFFFFB300), () => _run(UserData.toggleList(it))),
          _action(Icons.download_rounded, Icons.download_rounded, 'Download', false, Colors.greenAccent, _showDownloadOptions),
        ],
      ),
    );
  }

  Widget _seasonRow() {
    return SizedBox(
      height: 40,
      child: ListView.separated(
        scrollDirection: Axis.horizontal,
        physics: const BouncingScrollPhysics(),
        itemCount: seasonCount,
        separatorBuilder: (_, __) => const SizedBox(width: 8),
        itemBuilder: (_, i) {
          final s = i + 1;
          final sel = s == season;
          return GestureDetector(
            onTap: () {
              if (sel) return;
              setState(() {
                season = s;
                episode = 1;
                epNames = [];
              });
              _loadEpisodes();
              if (playing) UserData.addHistory(it, season: season, episode: episode);
            },
            child: AnimatedContainer(
              duration: const Duration(milliseconds: 220),
              padding: const EdgeInsets.symmetric(horizontal: 18),
              alignment: Alignment.center,
              decoration: BoxDecoration(
                gradient: sel ? kRedGradient : null,
                color: sel ? null : kCard,
                borderRadius: BorderRadius.circular(20),
                border: Border.all(color: sel ? Colors.transparent : Colors.white10),
              ),
              child: Text('Season $s', style: TextStyle(fontWeight: FontWeight.w700, color: sel ? Colors.white : Colors.white70)),
            ),
          );
        },
      ),
    );
  }

  void _pickEpisode(int e) {
    setState(() => episode = e);
    if (playing) UserData.addHistory(it, season: season, episode: episode);
  }

  List<Item> get related {
    final g = it['genres'];
    final mine = g is List ? g.map((e) => '$e').toSet() : <String>{};
    if (mine.isEmpty) return [];
    int score(Item e) {
      final x = e['genres'];
      return x is List ? x.where((a) => mine.contains('$a')).length : 0;
    }
    final l = Store.content.value.where((e) => e['id'] != it['id'] && score(e) > 0).toList();
    l.sort((a, b) => score(b).compareTo(score(a)));
    return l.take(12).toList();
  }

                          label: isTv ? 'Play S$season · E$episode' : 'Play Movie',
                          onTap: _play,
                        ),
                      ),
                      const SizedBox(width: 10),
                      Expanded(
                        child: OutlinedButton.icon(
                          onPressed: _showDownloadOptions,
                          icon: const Icon(Icons.download_rounded, color: Colors.greenAccent),
                          label: const Text('Download', style: TextStyle(color: Colors.white)),
                          style: OutlinedButton.styleFrom(
                            side: const BorderSide(color: Colors.greenAccent),
                            shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(16)),
                            padding: const EdgeInsets.symmetric(vertical: 14),
                          ),
                        ),
                      ),
                    ],
                  ),
                  if ('${it['overview'] ?? ''}'.isNotEmpty) ...[
                    const SizedBox(height: 20),
                    const Text('Overview', style: TextStyle(fontWeight: FontWeight.w800, fontSize: 15)),
                    const SizedBox(height: 6),
                    Text('${it['overview']}', style: const TextStyle(color: Colors.white70, height: 1.5)),
                  ],
                ],
              ),
            ),
            hRow(context, 'More like this', related),
          ],
        ),
      ),
    );
  }

  void _play() {
    if (servers.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('No server available for this title yet.')));
      return;
    }
    setState(() => playing = true);
    UserData.addHistory(it, season: season, episode: episode);
  }
}

class PlayerView extends StatefulWidget {
  final String url;
  const PlayerView({super.key, required this.url});

  @override
  State<PlayerView> createState() => _PlayerViewState();
}

class _PlayerViewState extends State<PlayerView> {
  late final WebViewController c;
  bool loading = true;
  bool loaded = false;

  String _root(String h) {
    final p = h.split('.');
    return p.length >= 2 ? p.sublist(p.length - 2).join('.') : h;
  }

  @override
  void initState() {
    super.initState();
    final base = _root(Uri.tryParse(widget.url)?.host ?? '');
    c = WebViewController()
      ..setJavaScriptMode(JavaScriptMode.unrestricted)
      ..setBackgroundColor(Colors.black)
      ..setNavigationDelegate(NavigationDelegate(
        onPageFinished: (_) {
          loaded = true;
          if (mounted) setState(() => loading = false);
        },
        onNavigationRequest: (r) {
          if (!loaded || !r.isMainFrame) return NavigationDecision.navigate;
          final u = Uri.tryParse(r.url);
          if (u == null) return NavigationDecision.prevent;
          if (u.scheme == 'about' || u.scheme == 'data' || _root(u.host) == base) {
            return NavigationDecision.navigate;
          }
          return NavigationDecision.prevent;
        },
      ));
    c.loadRequest(Uri.parse(widget.url));
  }

  @override
  Widget build(BuildContext context) {
    return Stack(
      fit: StackFit.expand,
      children: [
        WebViewWidget(controller: c),
        if (loading) const Center(child: CircularProgressIndicator(color: kRed)),
      ],
    );
  }
}
