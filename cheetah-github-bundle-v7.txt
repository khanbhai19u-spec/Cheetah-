import 'package:android_intent_plus/android_intent.dart';
import 'package:flutter/material.dart';
import 'package:flutter/services.dart';
import 'package:webview_flutter/webview_flutter.dart';
import 'package:url_launcher/url_launcher.dart';

import 'store.dart';
import 'theme.dart';
import 'userdata.dart';
import 'widgets.dart';

class DetailScreen extends StatefulWidget {
  final Item item;
  const DetailScreen({super.key, required this.item});

  @override
  State<DetailScreen> createState() => _DetailScreenState();
}

class _DetailScreenState extends State<DetailScreen> {
  Item get it => widget.item;
  Map<String, dynamic>? omdb;
  int season = 1, episode = 1, seasonCount = 1, epCount = 1, serverIdx = 0;
  List<String> epNames = [];
  bool playing = false;

  bool get isTv => it['type'] == 'tv';
  String get tmdbId => '${it['tmdbId'] ?? ''}';

  @override
  void initState() {
    super.initState();
    final h = UserData.history.value['${it['id']}'];
    if (h != null && isTv) {
      season = (h['season'] as num?)?.toInt() ?? 1;
      episode = (h['episode'] as num?)?.toInt() ?? 1;
    }
    _loadExtra();
  }

  Future<void> _loadExtra() async {
    if (tmdbId.isEmpty) return;
    String imdb = '';
    if (isTv) {
      final d = await Api.tmdb('tv/$tmdbId', {'append_to_response': 'external_ids'});
      if (d == null) return;
      imdb = '${d['external_ids']?['imdb_id'] ?? ''}';
      final n = (d['number_of_seasons'] as num?)?.toInt() ?? 1;
      if (mounted) setState(() => seasonCount = n < 1 ? 1 : n);
      _loadEpisodes();
    } else {
      final d = await Api.tmdb('movie/$tmdbId');
      imdb = '${d?['imdb_id'] ?? ''}';
    }
    final o = await Api.omdb(imdb);
    if (mounted && o != null) setState(() => omdb = o);
  }

  Future<void> _loadEpisodes() async {
    final d = await Api.tmdb('tv/$tmdbId/season/$season');
    final eps = d?['episodes'];
    if (eps is List && mounted) {
      setState(() {
        epCount = eps.isEmpty ? 1 : eps.length;
        epNames = [for (final e in eps) '${e is Map ? (e['name'] ?? '') : ''}'];
        if (episode > epCount) episode = 1;
      });
    }
  }

  String _fill(String tpl) => tpl
      .replaceAll('{tmdb_id}', tmdbId)
      .replaceAll('{season}', '$season')
      .replaceAll('{episode}', '$episode');

  List> get servers {
    final l = >[];
    if (tmdbId.isNotEmpty) {
      for (final s in Store.servers.value.where((s) => s['active'] != false)) {
        final tpl = '${(isTv ? s['tvUrl'] : s['movieUrl']) ?? ''}';
        if (tpl.isEmpty) continue;
        l.add({'name': '${s['name'] ?? 'Server'}', 'url': _fill(tpl)});
      }
    }
    final own = it['servers'];
    if (own is List) {
      for (final m in own) {
        if (m is Map && '${m['url'] ?? ''}'.isNotEmpty) {
          l.add({'name': '${m['name'] ?? 'Server'}', 'url': _fill('${m['url']}')});
        }
      }
    }
    return l;
  }

  // ডাইরেক্ট ডাউনলোড লিংক হ্যান্ডেল করার ফাংশন
  Future<void> _downloadMovie() async {
    final downloadLinks = it['download_links'];
    String targetUrl = '';
    
    if (downloadLinks is List && downloadLinks.isNotEmpty) {
      targetUrl = '${downloadLinks[0]['url'] ?? ''}';
    }
    
    if (targetUrl.isEmpty && servers.isNotEmpty) {
      targetUrl = servers[serverIdx]['url']!;
    }

    if (targetUrl.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('No download link available for this title.'))
      );
      return;
    }

    final uri = Uri.parse(targetUrl);
    if (await canLaunchUrl(uri)) {
      await launchUrl(uri, mode: LaunchMode.externalApplication);
    } else {
      if (mounted) {
        ScaffoldMessenger.of(context).showSnackBar(
          const SnackBar(content: Text('Could not launch download link.'))
        );
      }
    }
  }

  Widget _pill(String t, {IconData? icon, Color? color}) => Container(
        padding: const EdgeInsets.symmetric(horizontal: 10, vertical: 5),
        decoration: BoxDecoration(
          color: kCard,
          borderRadius: BorderRadius.circular(20),
          border: Border.all(color: Colors.white10),
        ),
        child: Row(
          mainAxisSize: MainAxisSize.min,
          children: [
            if (icon != null) ...[Icon(icon, size: 14, color: color), const SizedBox(width: 4)],
            Text(t, style: const TextStyle(fontSize: 12, color: Colors.white70)),
          ],
        ),
      );

  Future<void> _run(Future<void> f) async {
    try {
      await f.timeout(const Duration(seconds: 8));
    } catch (e) {
      if (!mounted) return;
      final t = '$e';
      ScaffoldMessenger.of(context).showSnackBar(SnackBar(
        content: Text('Could not save: ${t.length > 90 ? t.substring(0, 90) : t}'),
      ));
    }
  }

  Widget _action(IconData on, IconData off, String label, bool active, Color activeColor, VoidCallback tap) {
    return Expanded(
      child: GestureDetector(
        behavior: HitTestBehavior.opaque,
        onTap: () {
          HapticFeedback.lightImpact();
          tap();
        },
        child: Column(
          children: [
            AnimatedContainer(
              duration: const Duration(milliseconds: 220),
              width: 52,
              height: 52,
              decoration: BoxDecoration(
                color: active ? activeColor.withOpacity(0.18) : kCard,
                shape: BoxShape.circle,
                border: Border.all(color: active ? activeColor : Colors.white10, width: 1.4),
              ),
              child: AnimatedSwitcher(
                duration: const Duration(milliseconds: 200),
                child: Icon(active ? on : off, key: ValueKey(active), color: active ? activeColor : Colors.white70, size: 24),
              ),
            ),
            const SizedBox(height: 6),
            Text(label, style: TextStyle(fontSize: 12, color: active ? activeColor : Colors.white60, fontWeight: FontWeight.w600)),
          ],
        ),
      ),
    );
  }

  Widget _actions() {
    return AnimatedBuilder(
      animation: Listenable.merge([UserData.likes, UserData.favorites, UserData.watchlist]),
      builder: (_, __) => Row(
        children: [
          _action(Icons.thumb_up_rounded, Icons.thumb_up_outlined, 'Like', UserData.isLiked(it), const Color(0xFF29B6F6), () => _run(UserData.toggleLike(it))),
          _action(Icons.favorite_rounded, Icons.favorite_border_rounded, 'Favorite', UserData.isFav(it), kRedLight, () => _run(UserData.toggleFav(it))),
          _action(Icons.bookmark_rounded, Icons.bookmark_border_rounded, 'Watchlist', UserData.inList(it), const Color(0xFFFFB300), () => _run(UserData.toggleList(it))),
          _action(Icons.download_rounded, Icons.download_rounded, 'Download', false, Colors.greenAccent, _downloadMovie),
        ],
      ),
    );
  }

  Widget _seasonRow() {
    return SizedBox(
      height: 40,
      child: ListView.separated(
        scrollDirection: Axis.horizontal,
        physics: const BouncingScrollPhysics(),
        itemCount: seasonCount,
        separatorBuilder: (_, __) => const SizedBox(width: 8),
        itemBuilder: (_, i) {
          final s = i + 1;
          final sel = s == season;
          return GestureDetector(
            onTap: () {
              if (sel) return;
              setState(() {
                season = s;
                episode = 1;
                epNames = [];
              });
              _loadEpisodes();
              if (playing) UserData.addHistory(it, season: season, episode: episode);
            },
            child: AnimatedContainer(
              duration: const Duration(milliseconds: 220),
              padding: const EdgeInsets.symmetric(horizontal: 18),
              alignment: Alignment.center,
              decoration: BoxDecoration(
                gradient: sel ? kRedGradient : null,
                color: sel ? null : kCard,
                borderRadius: BorderRadius.circular(20),
                border: Border.all(color: sel ? Colors.transparent : Colors.white10),
              ),
              child: Text('Season $s', style: TextStyle(fontWeight: FontWeight.w700, color: sel ? Colors.white : Colors.white70)),
            ),
          );
        },
      ),
    );
  }

  void _pickEpisode(int e) {
    setState(() => episode = e);
    if (playing) UserData.addHistory(it, season: season, episode: episode);
  }

  Widget _episodeRow() {
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Row(
          children: [
            Expanded(child: Text('Episodes · $epCount', style: const TextStyle(fontWeight: FontWeight.w800, fontSize: 15))),
          ],
        ),
        const SizedBox(height: 8),
        SizedBox(
          height: 74,
          child: ListView.separated(
            scrollDirection: Axis.horizontal,
            physics: const BouncingScrollPhysics(),
            itemCount: epCount,
            separatorBuilder: (_, __) => const SizedBox(width: 10),
            itemBuilder: (_, i) {
              final e = i + 1;
              final sel = e == episode;
              final name = i < epNames.length ? epNames[i] : '';
              return GestureDetector(
                onTap: () => _pickEpisode(e),
                child: AnimatedContainer(
                  duration: const Duration(milliseconds: 220),
                  width: 132,
                  padding: const EdgeInsets.all(12),
                  decoration: BoxDecoration(
                    color: sel ? kRed.withOpacity(0.16) : kCard,
                    borderRadius: BorderRadius.circular(16),
                    border: Border.all(color: sel ? kRed : Colors.white10, width: sel ? 1.6 : 1),
                  ),
                  child: Column(
                    crossAxisAlignment: CrossAxisAlignment.start,
                    mainAxisAlignment: MainAxisAlignment.center,
                    children: [
                      Row(children: [
                        if (sel) const Padding(
                          padding: EdgeInsets.only(right: 4),
                          child: Icon(Icons.play_arrow_rounded, size: 18, color: kRedLight),
                        ),
                        Text('EP $e', style: TextStyle(fontWeight: FontWeight.w800, color: sel ? kRedLight : Colors.white, fontSize: 14)),
                      ]),
                      if (name.isNotEmpty) ...[
                        const SizedBox(height: 3),
                        Text(name, maxLines: 1, overflow: TextOverflow.ellipsis, style: const TextStyle(fontSize: 11.5, color: Colors.white54)),
                      ],
                    ],
                  ),
                ),
              );
            },
          ),
        ),
      ],
    );
  }

  List<Item> get related {
    final g = it['genres'];
    final mine = g is List ? g.map((e) => '$e').toSet() : <String>{};
    if (mine.isEmpty) return [];
    int score(Item e) {
      final x = e['genres'];
      return x is List ? x.where((a) => mine.contains('$a')).length : 0;
    }
    final l = Store.content.value.where((e) => e['id'] != it['id'] && score(e) > 0).toList();
    l.sort((a, b) => score(b).compareTo(score(a)));
    return l.take(12).toList();
  }

  @override
  Widget build(BuildContext context) {
    final sv = servers;
    final idx = sv.isEmpty ? 0 : (serverIdx >= sv.length ? 0 : serverIdx);
    final url = sv.isEmpty ? '' : sv[idx]['url']!;
    final g = it['genres'];
    final genres = g is List ? g.map((e) => '$e').toList() : <String>[];
    final backdrop = '${it['backdrop'] ?? ''}'.isNotEmpty ? '${it['backdrop']}' : '${it['poster'] ?? ''}';
    final rating = it['rating'] is num ? (it['rating'] as num).toDouble() : 0.0;
    final imdb = '${omdb?['imdbRating'] ?? ''}';

    return Scaffold(
      body: SafeArea(
        bottom: false,
        child: ListView(
          physics: const BouncingScrollPhysics(),
          padding: const EdgeInsets.only(bottom: 32),
          children: [
            AspectRatio(
              aspectRatio: 16 / 9,
              child: Stack(
                fit: StackFit.expand,
                children: [
                  if (playing && url.isNotEmpty)
                    PlayerView(key: ValueKey(url), url: url)
                  else ...[
                    Img(backdrop),
                    const DecoratedBox(
                      decoration: BoxDecoration(
                        gradient: LinearGradient(
                          begin: Alignment.topCenter,
                          end: Alignment.bottomCenter,
                          colors: [Color(0x66000000), Colors.transparent, kBg],
                          stops: [0, 0.4, 1],
                        ),
                      ),
                    ),
                    Center(
                      child: GestureDetector(
                        onTap: _play,
                        child: Container(
                          width: 68,
                          height: 68,
                          decoration: BoxDecoration(
                            gradient: kRedGradient,
                            shape: BoxShape.circle,
                            border: Border.all(color: Colors.white24, width: 2),
                            boxShadow: [BoxShadow(color: kRed.withOpacity(0.55), blurRadius: 28)],
                          ),
                          child: const Icon(Icons.play_arrow_rounded, size: 42),
                        ),
                      ),
                    ),
                  ],
                  Positioned(
                    top: 8,
                    left: 8,
                    child: Material(
                      color: Colors.black54,
                      shape: const CircleBorder(),
                      child: IconButton(
                        icon: const Icon(Icons.arrow_back_rounded),
                        onPressed: () => Navigator.pop(context),
                      ),
                    ),
                  ),
                ],
              ),
            ),
            Padding(
              padding: const EdgeInsets.fromLTRB(16, 14, 16, 0),
              child: Column(
                crossAxisAlignment: CrossAxisAlignment.start,
                children: [
                  Text('${it['title'] ?? ''}', style: const TextStyle(fontSize: 24, fontWeight: FontWeight.w800, height: 1.15)),
                  const SizedBox(height: 12),
                  Wrap(
                    spacing: 8,
                    runSpacing: 8,
                    children: [
                      if ('${it['year'] ?? ''}'.isNotEmpty) _pill('${it['year']}'),
                      if (rating > 0) _pill('TMDB ${rating.toStringAsFixed(1)}', icon: Icons.star_rounded, color: Colors.amber),
                      if (imdb.isNotEmpty && imdb != 'N/A') _pill('IMDb $imdb', icon: Icons.star_rounded, color: Colors.amber),
                      for (final x in genres) _pill(x),
                    ],
                  ),
                  const SizedBox(height: 18),
                  _actions(),
                  if (isTv && tmdbId.isNotEmpty) ...[
                    const SizedBox(height: 20),
                    _seasonRow(),
                    const SizedBox(height: 14),
                    _episodeRow(),
                  ],
                  const SizedBox(height: 18),
                  if (sv.isNotEmpty) ...[
                    const Text('Servers', style: TextStyle(fontWeight: FontWeight.w800, fontSize: 15)),
                    const SizedBox(height: 8),
                    Wrap(
                      spacing: 8,
                      runSpacing: 4,
                      children: [
                        for (var i = 0; i < sv.length; i++)
                          ChoiceChip(
                            avatar: Icon(Icons.dns_rounded, size: 16, color: i == idx ? Colors.white : Colors.white54),
                            label: Text(sv[i]['name']!),
                            selected: i == idx,
                            showCheckmark: false,
                            selectedColor: kRed,
                            backgroundColor: kCard,
                            side: BorderSide.none,
                            shape: const StadiumBorder(),
                            onSelected: (_) => setState(() => serverIdx = i),
                          ),
                      ],
                    ),
                  ],
                  const SizedBox(height: 16),
                  Row(
                    children: [
                      Expanded(
                        child: GradientButton(
                          icon: Icons.play_arrow_rounded,
                          label: isTv ? 'Play S$season · E$episode' : 'Play Movie',
                          onTap: _play,
                        ),
                      ),
                      const SizedBox(width: 10),
                      Expanded(
                        child: OutlinedButton.icon(
                          onPressed: _downloadMovie,
                          icon: const Icon(Icons.download_rounded, color: Colors.greenAccent),
                          label: const Text('Download', style: TextStyle(color: Colors.white)),
                          style: OutlinedButton.styleFrom(
                            side: const BorderSide(color: Colors.greenAccent),
                            shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(16)),
                            padding: const EdgeInsets.symmetric(vertical: 14),
                          ),
                        ),
                      ),
                    ],
                  ),
                  if ('${it['overview'] ?? ''}'.isNotEmpty) ...[
                    const SizedBox(height: 20),
                    const Text('Overview', style: TextStyle(fontWeight: FontWeight.w800, fontSize: 15)),
                    const SizedBox(height: 6),
                    Text('${it['overview']}', style: const TextStyle(color: Colors.white70, height: 1.5)),
                  ],
                ],
              ),
            ),
            hRow(context, 'More like this', related),
          ],
        ),
      ),
    );
  }

  void _play() {
    if (servers.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('No server available for this title yet.')));
      return;
    }
    setState(() => playing = true);
    UserData.addHistory(it, season: season, episode: episode);
  }
}

class PlayerView extends StatefulWidget {
  final String url;
  const PlayerView({super.key, required this.url});

  @override
  State<PlayerView> createState() => _PlayerViewState();
}

class _PlayerViewState extends State<PlayerView> {
  late final WebViewController c;
  bool loading = true;
  bool loaded = false;

  String _root(String h) {
    final p = h.split('.');
    return p.length >= 2 ? p.sublist(p.length - 2).join('.') : h;
  }

  @override
  void initState() {
    super.initState();
    final base = _root(Uri.tryParse(widget.url)?.host ?? '');
    c = WebViewController()
      ..setJavaScriptMode(JavaScriptMode.unrestricted)
      ..setBackgroundColor(Colors.black)
      ..setNavigationDelegate(NavigationDelegate(
        onPageFinished: (_) {
          loaded = true;
          if (mounted) setState(() => loading = false);
        },
        onNavigationRequest: (r) {
          if (!loaded || !r.isMainFrame) return NavigationDecision.navigate;
          final u = Uri.tryParse(r.url);
          if (u == null) return NavigationDecision.prevent;
          if (u.scheme == 'about' || u.scheme == 'data' || _root(u.host) == base) {
            return NavigationDecision.navigate;
          }
          return NavigationDecision.prevent;
        },
      ));
    c.loadRequest(Uri.parse(widget.url));
  }

  @override
  Widget build(BuildContext context) {
    return Stack(
      fit: StackFit.expand,
      children: [
        WebViewWidget(controller: c),
        if (loading) const Center(child: CircularProgressIndicator(color: kRed)),
      ],
    );
  }
}
