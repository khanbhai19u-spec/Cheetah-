/// TMDB / OMDB access. Keys are stored in Firestore (config/api) and set
/// from the admin panel, so they never live in the source code.
class Api {
  static String tmdbKey = '';
  static String omdbKey = '';

  static Future<void> load() async {
    try {
      final d = await db.collection('config').doc('api').get();
      final m = d.data() ?? {};
      tmdbKey = '${m['tmdbKey'] ?? ''}'.trim();
      omdbKey = '${m['omdbKey'] ?? ''}'.trim();
    } catch (_) {}
  }

  static Future<Map<String, dynamic>?> tmdb(String path,
      [Map<String, String> q = const {}]) async {
    if (tmdbKey.isEmpty) return null;
    try {
      final uri =
          Uri.https('api.themoviedb.org', '/3/$path', {'api_key': tmdbKey, ...q});
      final r = await http.get(uri).timeout(const Duration(seconds: 15));
      if (r.statusCode == 200) {
        return Map<String, dynamic>.from(jsonDecode(r.body) as Map);
      }
    } catch (_) {}
    return null;
  }

  static Future<Map<String, dynamic>?> omdb(String imdbId) async {
    if (omdbKey.isEmpty || imdbId.isEmpty) return null;
    try {
      final uri = Uri.https('www.omdbapi.com', '/', {'apikey': omdbKey, 'i': imdbId});
      final r = await http.get(uri).timeout(const Duration(seconds: 15));
      if (r.statusCode == 200) {
        final j = Map<String, dynamic>.from(jsonDecode(r.body) as Map);
        if (j['Response'] == 'True') return j;
      }
    } catch (_) {}
    return null;
  }
}

/// Live copies of the Firestore collections the app needs.
class Store {
  static final content = ValueNotifier<List<Item>>([]);
  static final slides = ValueNotifier<List<Item>>([]);
  static final ads = ValueNotifier<List<Item>>([]);
  static final servers = ValueNotifier<List<Item>>([]);
  static final popups = ValueNotifier<List<Item>>([]);
  static final notifs = ValueNotifier<List<Item>>([]);

  static final List<StreamSubscription> _subs = [];

  static int _ts(Item m) {
    final t = m['createdAt'];
    return t is Timestamp ? t.millisecondsSinceEpoch : 0;
  }

  static num _n(dynamic v) => v is num ? v : 0;

  static int _newest(Item a, Item b) => _ts(b).compareTo(_ts(a));

  static void start() {
    if (_subs.isNotEmpty) return;
    _listen('content', content, _newest);
    _listen('slides', slides, (a, b) => _n(a['order']).compareTo(_n(b['order'])));
    _listen('ads', ads, _newest);
    _listen('servers', servers, (a, b) => _n(a['order']).compareTo(_n(b['order'])));
    _listen('popups', popups, _newest);
    _listen('notifications', notifs, _newest);
  }

  static void stop() {
    for (final s in _subs) {
      s.cancel();
    }
    _subs.clear();
  }

  static void _listen(String col, ValueNotifier<List<Item>> n, int Function(Item, Item) cmp) {
    _subs.add(db.collection(col).snapshots().listen((s) {
      final l = s.docs.map((d) => <String, dynamic>{...d.data(), 'id': d.id}).toList();
      l.sort(cmp);
      n.value = l;
    }, onError: (_) {}));
  }
}

Future<void> openLink(String url) async {
  if (url.trim().isEmpty) return;
  try {
    final u = Uri.tryParse(url.trim());
    if (u != null) await launchUrl(u, mode: LaunchMode.externalApplication);
  } catch (_) {}
}

import 'package:firebase_auth/firebase_auth.dart';

class AuthScreen extends StatefulWidget {
  const AuthScreen({super.key});

  @override
  State<AuthScreen> createState() => _AuthScreenState();
}

class _AuthScreenState extends State<AuthScreen> {
  bool login = true;
  bool busy = false;
  bool hide = true;
  String? err;
  final name = TextEditingController();
  final email = TextEditingController();
  final pass = TextEditingController();

  @override
  void dispose() {
    name.dispose();
    email.dispose();
    pass.dispose();
    super.dispose();
  }

  String _msg(FirebaseAuthException e) {
    switch (e.code) {
      case 'invalid-email':
        return 'Please enter a valid email address.';
      case 'user-not-found':
      case 'wrong-password':
      case 'invalid-credential':
        return 'Incorrect email or password.';
      case 'email-already-in-use':
        return 'An account with this email already exists.';
      case 'weak-password':
        return 'Password must be at least 6 characters.';
      case 'network-request-failed':
        return 'No internet connection.';
      case 'too-many-requests':
        return 'Too many attempts. Try again later.';
      default:
        return e.message ?? 'Something went wrong.';
    }
  }

  Future<void> submit() async {
    FocusScope.of(context).unfocus();
    setState(() {
      busy = true;
      err = null;
    });
    try {
      final auth = FirebaseAuth.instance;
      if (login) {
        await auth.signInWithEmailAndPassword(email: email.text.trim(), password: pass.text);
      } else {
        if (name.text.trim().isEmpty) {
          throw FirebaseAuthException(code: 'custom', message: 'Please enter your name.');
        }
        final cr = await auth.createUserWithEmailAndPassword(
            email: email.text.trim(), password: pass.text);
        await cr.user?.updateDisplayName(name.text.trim());
        try {
          await db.collection('users').doc(cr.user!.uid).set({
            'name': name.text.trim(),
            'email': email.text.trim(),
            'createdAt': FieldValue.serverTimestamp(),
          });
        } catch (_) {}
      }
    } on FirebaseAuthException catch (e) {
      err = _msg(e);
    } catch (e) {
      err = 'Something went wrong. Try again.';
    }
    if (mounted) setState(() => busy = false);
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: SafeArea(
        child: Center(
          child: SingleChildScrollView(
            padding: const EdgeInsets.all(24),
            child: ConstrainedBox(
              constraints: const BoxConstraints(maxWidth: 420),
              child: Column(
                children: [
                  ClipRRect(
                    borderRadius: BorderRadius.circular(24),
                    child: Image.asset('assets/icon.png', width: 96, height: 96),
                  ),
                  const SizedBox(height: 14),
                  const Text('Cheetah',
                      style: TextStyle(fontSize: 30, fontWeight: FontWeight.w800, letterSpacing: 0.5)),
                  const SizedBox(height: 4),
                  Text(login ? 'Welcome back' : 'Create your account',
                      style: const TextStyle(color: Colors.white54)),
                  const SizedBox(height: 28),
                  if (!login) ...[
                    _field(name, 'Full name', Icons.person_outline),
                    const SizedBox(height: 12),
                  ],
                  _field(email, 'Email', Icons.mail_outline, type: TextInputType.emailAddress),
                  const SizedBox(height: 12),
                  TextField(
                    controller: pass,
                    obscureText: hide,
                    decoration: _dec('Password', Icons.lock_outline).copyWith(
                      suffixIcon: IconButton(
                        icon: Icon(hide ? Icons.visibility_off_outlined : Icons.visibility_outlined),
                        onPressed: () => setState(() => hide = !hide),
                      ),
                    ),
                  ),
                  if (err != null) ...[
                    const SizedBox(height: 12),
                    Text(err!, style: const TextStyle(color: Colors.redAccent), textAlign: TextAlign.center),
                  ],
                  const SizedBox(height: 20),
                  SizedBox(
                    width: double.infinity,
                    height: 50,
                    child: FilledButton(
                      style: FilledButton.styleFrom(
                        backgroundColor: kRed,
                        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(14)),
                      ),
                      onPressed: busy ? null : submit,
                      child: busy
                          ? const SizedBox(
                              width: 22, height: 22, child: CircularProgressIndicator(strokeWidth: 2.5))
                          : Text(login ? 'Log in' : 'Sign up',
                              style: const TextStyle(fontSize: 16, fontWeight: FontWeight.w700)),
                    ),
                  ),
                  const SizedBox(height: 14),
                  TextButton(
                    onPressed: () => setState(() {
                      login = !login;
                      err = null;
                    }),
                    child: Text(login ? "Don't have an account? Sign up" : 'Already have an account? Log in'),
                  ),
                ],
              ),
            ),
          ),
        ),
      ),
    );
  }

  Widget _field(TextEditingController c, String label, IconData icon, {TextInputType? type}) =>
      TextField(controller: c, keyboardType: type, decoration: _dec(label, icon));

  InputDecoration _dec(String label, IconData icon) => InputDecoration(
        labelText: label,
        prefixIcon: Icon(icon),
        filled: true,
        fillColor: kCard,
        border: OutlineInputBorder(borderRadius: BorderRadius.circular(14), borderSide: BorderSide.none),
      );
}

/// Runs after login: starts live data listeners, loads API keys, sets up push.
class Boot extends StatefulWidget {
  const Boot({super.key});

  @override
  State<Boot> createState() => _BootState();
}

class _BootState extends State<Boot> {
  late final Future<void> _f = _init();

  Future<void> _init() async {
    Store.start();
    await Api.load();
    try {
      final m = FirebaseMessaging.instance;
      await m.requestPermission();
      await m.subscribeToTopic('all');
    } catch (_) {}
  }

  @override
  void dispose() {
    Store.stop();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return FutureBuilder<void>(
      future: _f,
      builder: (_, s) => s.connectionState == ConnectionState.done ? const Shell() : const Splash(),
    );
  }
}

class Shell extends StatefulWidget {
  const Shell({super.key});

  @override
  State<Shell> createState() => _ShellState();
}

class _ShellState extends State<Shell> {
  int tab = 0;
  bool _popupDone = false;

  @override
  void initState() {
    super.initState();
    Store.popups.addListener(_checkPopup);
    WidgetsBinding.instance.addPostFrameCallback((_) => _checkPopup());
  }

  @override
  void dispose() {
    Store.popups.removeListener(_checkPopup);
    super.dispose();
  }

  Future<void> _checkPopup() async {
    if (_popupDone || !mounted) return;
    final list = Store.popups.value.where((p) => p['active'] == true).toList();
    if (list.isEmpty) return;
    _popupDone = true;
    final p = list.first;
    final always = p['showAlways'] == true;
    final sp = await SharedPreferences.getInstance();
    final seen = sp.getStringList('seen_popups') ?? <String>[];
    if (!always && seen.contains('${p['id']}')) return;
    if (!mounted) return;
    await showDialog(
      context: context,
      builder: (ctx) => Dialog(
        backgroundColor: kCard,
        shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(20)),
        clipBehavior: Clip.antiAlias,
        child: SingleChildScrollView(
          child: Column(
            mainAxisSize: MainAxisSize.min,
            children: [
              if ('${p['image'] ?? ''}'.isNotEmpty)
                AspectRatio(aspectRatio: 16 / 9, child: Img(p['image']?.toString())),
              Padding(
                padding: const EdgeInsets.all(18),
                child: Column(
                  children: [
                    if ('${p['title'] ?? ''}'.isNotEmpty)
                      Text('${p['title']}',
                          textAlign: TextAlign.center,
                          style: const TextStyle(fontSize: 18, fontWeight: FontWeight.w800)),
                    if ('${p['message'] ?? ''}'.isNotEmpty) ...[
                      const SizedBox(height: 8),
                      Text('${p['message']}',
                          textAlign: TextAlign.center, style: const TextStyle(color: Colors.white70)),
                    ],
                    const SizedBox(height: 16),
                    Row(
                      children: [
                        Expanded(
                          child: OutlinedButton(
                            onPressed: () => Navigator.pop(ctx),
                            child: const Text('Close'),
                          ),
                        ),
                        if ('${p['link'] ?? ''}'.isNotEmpty) ...[
                          const SizedBox(width: 10),
                          Expanded(
                            child: FilledButton(
                              style: FilledButton.styleFrom(backgroundColor: kRed),
                              onPressed: () {
                                Navigator.pop(ctx);
                                openLink('${p['link']}');
                              },
                              child: Text('${p['buttonText'] ?? ''}'.isEmpty ? 'Open' : '${p['buttonText']}'),
                            ),
                          ),
                        ],
                      ],
                    ),
                  ],
                ),
              ),
            ],
          ),
        ),
      ),
    );
    if (!always) {
      seen.add('${p['id']}');
      await sp.setStringList('seen_popups', seen);
    }
  }

  @override
  Widget build(BuildContext context) {
    const pages = [HomeScreen(), CategoryScreen(), SearchScreen(), ProfileScreen()];
    return Scaffold(
      body: IndexedStack(index: tab, children: pages),
      bottomNavigationBar: NavigationBar(
        selectedIndex: tab,
        onDestinationSelected: (i) => setState(() => tab = i),
        destinations: const [
          NavigationDestination(icon: Icon(Icons.home_outlined), selectedIcon: Icon(Icons.home), label: 'Home'),

class HeroSlider extends StatefulWidget {
  const HeroSlider({super.key});

  @override
  State<HeroSlider> createState() => _HeroSliderState();

class _HeroSliderState extends State<HeroSlider> {
  final _pc = PageController(viewportFraction: 0.92);
  Timer? _t;
  int _i = 0;

  @override
  void initState() {
    super.initState();
    _t = Timer.periodic(const Duration(seconds: 4), (_) {
      if (!_pc.hasClients) return;
      final n = _items().length;
      if (n < 2) return;
      _pc.animateToPage((_i + 1) % n,
          duration: const Duration(milliseconds: 450), curve: Curves.easeInOut);
    });
  }

  @override
  void dispose() {
    _t?.cancel();
    _pc.dispose();
    super.dispose();
  }

  List<_Slide> _items() {
    final s = Store.slides.value
        .where((e) => e['active'] != false && '${e['image'] ?? ''}'.isNotEmpty)
        .toList();
    if (s.isNotEmpty) {
      return s
          .map((e) => _Slide('${e['image']}', '${e['title'] ?? ''}', '${e['contentId'] ?? ''}',
              '${e['link'] ?? ''}'))
          .toList();
    }
    return Store.content.value
        .where((e) => '${e['backdrop'] ?? ''}'.isNotEmpty)
        .take(5)
        .map((e) => _Slide('${e['backdrop']}', '${e['title'] ?? ''}', '${e['id']}', ''))
        .toList();
  }

  void _tap(_Slide s) {
    if (s.contentId.isNotEmpty) {
      final it = Store.content.value.where((e) => e['id'] == s.contentId).firstOrNull;
      if (it != null) {
        openItem(context, it);
        return;
      }
    }
    openLink(s.link);
  }

  @override
  Widget build(BuildContext context) {
    return AnimatedBuilder(
      animation: Listenable.merge([Store.slides, Store.content]),
      builder: (_, __) {
        final items = _items();
        if (items.isEmpty) return const SizedBox.shrink();
        return LayoutBuilder(
          builder: (_, box) {
            final h = box.maxWidth * 0.92 * 9 / 16;
            return Column(
              children: [
                SizedBox(
                  height: h,
                  child: PageView.builder(
                    controller: _pc,
                    itemCount: items.length,
                    onPageChanged: (i) => setState(() => _i = i),
                    itemBuilder: (_, i) => _slide(items[i]),
                  ),
                ),
                const SizedBox(height: 10),
                Row(
                  mainAxisAlignment: MainAxisAlignment.center,
                  children: [
                    for (var i = 0; i < items.length; i++)
                      AnimatedContainer(
                        duration: const Duration(milliseconds: 250),
                        margin: const EdgeInsets.symmetric(horizontal: 3),
                        width: i == _i ? 18 : 6,
                        height: 6,
                        decoration: BoxDecoration(
                          color: i == _i ? kRed : Colors.white24,
                          borderRadius: BorderRadius.circular(3),
                        ),
                      ),
                  ],
                ),
              ],
            );
          },
        );
      },
    );
  }

  Widget _slide(_Slide s) {
    return Padding(
      padding: const EdgeInsets.symmetric(horizontal: 4),
      child: GestureDetector(
        onTap: () => _tap(s),
        child: ClipRRect(
          borderRadius: BorderRadius.circular(18),
          child: Stack(
            fit: StackFit.expand,
            children: [
              Img(s.image),
              const DecoratedBox(
                decoration: BoxDecoration(
                  gradient: LinearGradient(
                    begin: Alignment.topCenter,
                    end: Alignment.bottomCenter,
                    colors: [Colors.transparent, Color(0xDD000000)],
                    stops: [0.45, 1],
                  ),
                ),
              ),
              if (s.title.isNotEmpty)
                Positioned(
                  left: 14,
                  right: 14,
                  bottom: 12,
                  child: Text(
                    s.title,
                    maxLines: 2,
                    overflow: TextOverflow.ellipsis,
                    style: const TextStyle(fontSize: 17, fontWeight: FontWeight.w800),
                  ),
                ),
            ],
          ),

class _DetailScreenState extends State<DetailScreen> {
  Item get it => widget.item;
  Map<String, dynamic>? omdb;
  int season = 1, episode = 1, seasonCount = 1, epCount = 1, serverIdx = 0;
  bool playing = false;

  bool get isTv => it['type'] == 'tv';
  String get tmdbId => '${it['tmdbId'] ?? ''}';

  @override
  void initState() {
    super.initState();
    _loadExtra();
  }

  Future<void> _loadExtra() async {
    if (tmdbId.isEmpty) return;
    String imdb = '';
    if (isTv) {
      final d = await Api.tmdb('tv/$tmdbId', {'append_to_response': 'external_ids'});
      if (d == null) return;
      imdb = '${d['external_ids']?['imdb_id'] ?? ''}';
      final n = (d['number_of_seasons'] as num?)?.toInt() ?? 1;
      if (mounted) setState(() => seasonCount = n < 1 ? 1 : n);
      _loadEpisodes();
    } else {
      final d = await Api.tmdb('movie/$tmdbId');
      imdb = '${d?['imdb_id'] ?? ''}';
    }
    final o = await Api.omdb(imdb);
    if (mounted && o != null) setState(() => omdb = o);
  }

  Future<void> _loadEpisodes() async {
    final d = await Api.tmdb('tv/$tmdbId/season/$season');
    final eps = d?['episodes'];
    if (eps is List && mounted) {
      setState(() {
        epCount = eps.isEmpty ? 1 : eps.length;
        if (episode > epCount) episode = 1;
      });

        if (tpl.isEmpty) continue;
        l.add({'name': '${s['name'] ?? 'Server'}', 'url': _fill(tpl)});
      }

    final own = it['servers'];
    if (own is List) {

                    Text('${it['overview']}', style: const TextStyle(color: Colors.white70, height: 1.5)),
                  ],
                ],
              ),
            ),
            const AdBanner('detail'),
          ],
        ),
      ),
    );
  }

  void _play() {
    if (servers.isEmpty) {
      ScaffoldMessenger.of(context)
          .showSnackBar(const SnackBar(content: Text('No server available for this title yet.')));
      return;
    }
    setState(() => playing = true);
  }
}

/// Embedded player. Blocks pop-up style redirects to other sites once the
/// player page has finished loading.
class PlayerView extends StatefulWidget {
  final String url;
  const PlayerView({super.key, required this.url});

  @override
  State<PlayerView> createState() => _PlayerViewState();
}

class _PlayerViewState extends State<PlayerView> {
  late final WebViewController c;
  bool loading = true;
  bool loaded = false;

  String _root(String h) {
    final p = h.split('.');
    return p.length >= 2 ? p.sublist(p.length - 2).join('.') : h;
  }

  @override
  void initState() {
    super.initState();
    final base = _root(Uri.tryParse(widget.url)?.host ?? '');
    c = WebViewController()
      ..setJavaScriptMode(JavaScriptMode.unrestricted)
      ..setBackgroundColor(Colors.black)
      ..setNavigationDelegate(NavigationDelegate(
        onPageFinished: (_) {
          loaded = true;
          if (mounted) setState(() => loading = false);
        },
        onNavigationRequest: (r) {
          if (!loaded || !r.isMainFrame) return NavigationDecision.navigate;
          final u = Uri.tryParse(r.url);
          if (u == null) return NavigationDecision.prevent;
          if (u.scheme == 'about' || u.scheme == 'data' || _root(u.host) == base) {
            return NavigationDecision.navigate;
          }
          return NavigationDecision.prevent;
        },
      ));
    final lower = widget.url.toLowerCase().split('?').first;
    if (lower.endsWith('.mp4') || lower.endsWith('.webm')) {
      c.loadHtmlString(
          '<html><body style="margin:0;background:#000"><video src="${widget.url}" controls autoplay playsinline style="width:100%;height:100%"></video></body></html>');
    } else {
      c.loadRequest(Uri.parse(widget.url));
    }
  }

  @override
  Widget build(BuildContext context) {
    return Stack(
      fit: StackFit.expand,
      children: [
        WebViewWidget(controller: c),
        if (loading) const Center(child: CircularProgressIndicator()),
      ],
    );
  }
}
