class UserData {
  static final likes = ValueNotifier<Map<String, Item>>({});
  static final favorites = ValueNotifier<Map<String, Item>>({});
  static final watchlist = ValueNotifier<Map<String, Item>>({});
  static final history = ValueNotifier<Map<String, Item>>({});

  static String? _uid;
  static final List<StreamSubscription> _subs = [];

  static Map<String, ValueNotifier<Map<String, Item>>> get _all => {
        'likes': likes,
        'favorites': favorites,
        'watchlist': watchlist,
        'history': history,
      };

  static CollectionReference<Map<String, dynamic>> _col(String n) =>
      db.collection('users').doc(_uid).collection(n);

  static void start(String uid) {
    if (_uid == uid && _subs.isNotEmpty) return;
    stop();
    _uid = uid;
    for (final n in _all.values) {
      n.value = {};
    }
    _all.forEach((name, n) {
      _subs.add(_col(name).snapshots().listen((s) {
        final m = <String, Item>{};
        for (final d in s.docs) {
          m[d.id] = <String, dynamic>{...d.data(), 'id': d.id};
        }
        n.value = m;
      }, onError: (_) {}));
    });
  }

  static void stop() {
    for (final s in _subs) {
      s.cancel();
    }
    _subs.clear();
    _uid = null;
  }

  static List<Item> sorted(Map<String, Item> m, String field) {
    final now = DateTime.now().millisecondsSinceEpoch;
    int ts(Item e) {
      final t = e[field];
      return t is Timestamp ? t.millisecondsSinceEpoch : now;
    }

    final l = m.values.toList()..sort((a, b) => ts(b).compareTo(ts(a)));
    return l;
  }

  static bool isLiked(Item it) => likes.value.containsKey('${it['id']}');
  static bool isFav(Item it) => favorites.value.containsKey('${it['id']}');
  static bool inList(Item it) => watchlist.value.containsKey('${it['id']}');

  static Future<void> _toggle(String name, ValueNotifier<Map<String, Item>> n, Item it) async {
    if (_uid == null) return;
    final id = '${it['id']}';
    final cur = Map<String, Item>.from(n.value);
    final ref = _col(name).doc(id);
    if (cur.containsKey(id)) {
      cur.remove(id);
      n.value = cur;
      await ref.delete();
    } else {
      cur[id] = it;
      n.value = cur;
      await ref.set({...it, 'savedAt': FieldValue.serverTimestamp()});
    }
  }

  static Future<void> toggleLike(Item it) => _toggle('likes', likes, it);
  static Future<void> toggleFav(Item it) => _toggle('favorites', favorites, it);
  static Future<void> toggleList(Item it) => _toggle('watchlist', watchlist, it);

  static void addHistory(Item it, {int season = 1, int episode = 1}) {
    if (_uid == null) return;
    final id = '${it['id']}';
    final cur = Map<String, Item>.from(history.value);
    cur[id] = {...it, 'season': season, 'episode': episode};
    history.value = cur;
    try {
      _col('history')
          .doc(id)
          .set({...it, 'season': season, 'episode': episode, 'watchedAt': FieldValue.serverTimestamp()})
          .catchError((_) {});
    } catch (_) {}
  }
}

class AuthScreen extends StatefulWidget {
  const AuthScreen({super.key});

  @override
  State<AuthScreen> createState() => _AuthScreenState();
}

class _AuthScreenState extends State<AuthScreen> {
  final _email = TextEditingController();
  final _pass = TextEditingController();
  bool signup = false, busy = false, hide = true;
  String? error;

  @override
  void dispose() {
    _email.dispose();
    _pass.dispose();
    super.dispose();
  }

  String _msg(String code) {
    switch (code) {
      case 'invalid-email':
        return 'Invalid email address.';
      case 'user-not-found':
      case 'wrong-password':
      case 'invalid-credential':
        return 'Wrong email or password.';
      case 'email-already-in-use':
        return 'This email is already registered.';
      case 'weak-password':
        return 'Password is too weak (min 6 characters).';
      case 'network-request-failed':
        return 'No internet connection.';
      case 'too-many-requests':
        return 'Too many attempts. Try again later.';
      default:
        return 'Authentication failed ($code).';
    }
  }

  Future<void> _submit() async {
    final e = _email.text.trim();
    final p = _pass.text;
    if (e.isEmpty || p.length < 6) {
      setState(() => error = 'Enter a valid email and a password of 6+ characters.');
      return;
    }
    setState(() {
      busy = true;
      error = null;
    });
    try {
      final a = FirebaseAuth.instance;
      if (signup) {
        await a.createUserWithEmailAndPassword(email: e, password: p);
      } else {
        await a.signInWithEmailAndPassword(email: e, password: p);
      }
    } on FirebaseAuthException catch (x) {
      if (mounted) setState(() => error = _msg(x.code));
    } catch (_) {
      if (mounted) setState(() => error = 'Something went wrong. Check your internet.');
    } finally {
      if (mounted) setState(() => busy = false);
    }
  }

  Future<void> _reset() async {
    final e = _email.text.trim();
    if (e.isEmpty) {
      setState(() => error = 'Type your email first.');
      return;
    }
    try {
      await FirebaseAuth.instance.sendPasswordResetEmail(email: e);
      if (mounted) setState(() => error = 'Password reset email sent.');
    } on FirebaseAuthException catch (x) {
      if (mounted) setState(() => error = _msg(x.code));
    }
  }

  InputDecoration _dec(String hint, IconData icon, {Widget? suffix}) => InputDecoration(
        hintText: hint,
        prefixIcon: Icon(icon),
        suffixIcon: suffix,
        filled: true,
        fillColor: kCard,
        border: OutlineInputBorder(borderRadius: BorderRadius.circular(16), borderSide: BorderSide.none),
      );

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      body: SafeArea(
        child: Center(
          child: SingleChildScrollView(
            padding: const EdgeInsets.all(24),
            child: Column(
              children: [
                const Logo(size: 88),
                const SizedBox(height: 14),
                const Text('Cheetah', style: TextStyle(fontSize: 28, fontWeight: FontWeight.w900)),
                const SizedBox(height: 4),
                Text(signup ? 'Create your account' : 'Welcome back',
                    style: const TextStyle(color: Colors.white54)),
                const SizedBox(height: 28),
                TextField(
                  controller: _email,
                  keyboardType: TextInputType.emailAddress,
                  decoration: _dec('Email', Icons.mail_outline_rounded),
                ),
                const SizedBox(height: 12),
                TextField(
                  controller: _pass,
                  obscureText: hide,
                  onSubmitted: (_) => _submit(),
                  decoration: _dec(
                    'Password',
                    Icons.lock_outline_rounded,
                    suffix: IconButton(
                      icon: Icon(hide ? Icons.visibility_off_rounded : Icons.visibility_rounded),
                      onPressed: () => setState(() => hide = !hide),
                    ),
                  ),
                ),
                if (error != null) ...[
                  const SizedBox(height: 12),
                  Text(error!, textAlign: TextAlign.center, style: const TextStyle(color: Colors.orangeAccent)),
                ],
                const SizedBox(height: 18),
                SizedBox(
                  width: double.infinity,
                  height: 50,
                  child: FilledButton(
                    onPressed: busy ? null : _submit,
                    style: FilledButton.styleFrom(
                      backgroundColor: kRed,
                      shape: RoundedRectangleBorder(borderRadius: BorderRadius.circular(16)),
                    ),
                    child: busy
                        ? const SizedBox(width: 22, height: 22, child: CircularProgressIndicator(strokeWidth: 2.5, color: Colors.white))
                        : Text(signup ? 'Sign up' : 'Sign in',
                            style: const TextStyle(fontSize: 16, fontWeight: FontWeight.w800, color: Colors.white)),
                  ),
                ),
                const SizedBox(height: 8),
                if (!signup) TextButton(onPressed: _reset, child: const Text('Forgot password?')),
                TextButton(
                  onPressed: () => setState(() {
                    signup = !signup;
                    error = null;
                  }),
                  child: Text(signup ? 'Already have an account? Sign in' : 'New here? Create account'),
                ),
              ],
            ),
          ),
        ),
      ),
    );
  }
}

class _WebPageState extends State<WebPage> {
  late final WebViewController c;
  bool loading = true;

  @override
  void initState() {
    super.initState();
    c = WebViewController()
      ..setJavaScriptMode(JavaScriptMode.unrestricted)
      ..setBackgroundColor(Colors.black)
      ..setNavigationDelegate(NavigationDelegate(
        onPageFinished: (_) {
          if (mounted) setState(() => loading = false);
        },
        onNavigationRequest: (r) {
          if (Downloads.isDirect(r.url)) {
            Downloads.start(widget.title, '', r.url);
            ScaffoldMessenger.of(context).showSnackBar(
              const SnackBar(content: Text('Download started (see Profile > Downloads)')),
            );
            return NavigationDecision.prevent;
          }
          return NavigationDecision.navigate;
        },
      ))
      ..loadRequest(Uri.parse(widget.url));
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: Text(widget.title, maxLines: 1, overflow: TextOverflow.ellipsis, style: const TextStyle(fontSize: 16)),
        actions: [
          IconButton(
            icon: const Icon(Icons.download_done_rounded),
            onPressed: () => Navigator.push(context, MaterialPageRoute(builder: (_) => const DownloadsScreen())),
          ),
        ],
      ),
      body: Stack(
        fit: StackFit.expand,
        children: [
          WebViewWidget(controller: c),
          if (loading) const Center(child: CircularProgressIndicator(color: kRed)),
        ],
      ),
    );
  }
}

class DlTask {
  final String id, title, quality, url;
  String path;
  int received;
  int total;
  String status; // downloading | done | failed | canceled
  bool cancel = false;
  http.Client? client;

  DlTask({
    required this.id,
    required this.title,
    required this.quality,
    required this.url,
    this.path = '',
    this.received = 0,
    this.total = 0,
    this.status = 'downloading',
  });

  double get progress => total > 0 ? (received / total).clamp(0.0, 1.0) : 0;

  Map<String, dynamic> toJson() => {
        'id': id,
        'title': title,
        'quality': quality,
        'url': url,
        'path': path,
        'size': received,
      };

  factory DlTask.fromJson(Map<String, dynamic> j) => DlTask(
        id: '${j['id']}',
        title: '${j['title']}',
        quality: '${j['quality'] ?? ''}',
        url: '${j['url'] ?? ''}',
        path: '${j['path'] ?? ''}',
        received: (j['size'] as num?)?.toInt() ?? 0,
        total: (j['size'] as num?)?.toInt() ?? 0,
        status: 'done',
      );
}

class Downloads {
  static final tasks = ValueNotifier<List<DlTask>>([]);
  static const _key = 'dl_done_v1';
  static const _exts = ['.mp4', '.mkv', '.webm', '.m4v', '.mov', '.avi'];

  static bool isDirect(String url) {
    final p = Uri.tryParse(url)?.path.toLowerCase() ?? '';
    return _exts.any(p.endsWith);
  }

  static String _ext(String url) {
    final p = Uri.tryParse(url)?.path.toLowerCase() ?? '';
    for (final e in _exts) {
      if (p.endsWith(e)) return e;
    }
    return '.mp4';
  }

  static void _notify() => tasks.value = List.of(tasks.value);

  static Future<void> load() async {
    try {
      final sp = await SharedPreferences.getInstance();
      final raw = sp.getString(_key);
      if (raw == null) return;
      final l = (jsonDecode(raw) as List)
          .map((e) => DlTask.fromJson(Map<String, dynamic>.from(e as Map)))
          .where((t) => File(t.path).existsSync())
          .toList();
      final running = tasks.value.where((t) => t.status == 'downloading').toList();
      tasks.value = [...running, ...l];
    } catch (_) {}
  }

  static Future<void> _save() async {
    try {
      final sp = await SharedPreferences.getInstance();
      final done = tasks.value.where((t) => t.status == 'done').map((t) => t.toJson()).toList();
      await sp.setString(_key, jsonEncode(done));
    } catch (_) {}
  }

  static Future<Directory> _dir() async {
    Directory? base;
    try {
      base = await getExternalStorageDirectory();
    } catch (_) {}
    base ??= await getApplicationDocumentsDirectory();
    final d = Directory('${base.path}/CheetahDownloads');
    if (!d.existsSync()) d.createSync(recursive: true);
    return d;
  }

  static DlTask start(String title, String quality, String url) {
    final id = DateTime.now().millisecondsSinceEpoch.toString();
    final t = DlTask(id: id, title: title, quality: quality, url: url);
    tasks.value = [t, ...tasks.value];
    unawaited(_run(t));
    return t;
  }

  static Future<void> _run(DlTask t) async {
    File? f;
    IOSink? sink;
    final client = http.Client();
    t.client = client;
    try {
      final dir = await _dir();
      final safe = t.title.replaceAll(RegExp(r'[^A-Za-z0-9 _-]'), '').trim().replaceAll(' ', '_');
      final q = t.quality.isEmpty ? '' : '${t.quality}_';
      t.path = '${dir.path}/${safe.isEmpty ? 'video' : safe}_$q${t.id}${_ext(t.url)}';
      f = File(t.path);
      sink = f.openWrite();
      final res = await client.send(http.Request('GET', Uri.parse(t.url))).timeout(const Duration(seconds: 30));
      if (res.statusCode != 200) throw 'HTTP ${res.statusCode}';
      t.total = res.contentLength ?? 0;
      var last = DateTime.now();
      await for (final chunk in res.stream) {
        if (t.cancel) throw 'cancelled';
        sink.add(chunk);
        t.received += chunk.length;
        if (DateTime.now().difference(last).inMilliseconds > 300) {
          last = DateTime.now();
          _notify();
        }
      }
      await sink.flush();
      await sink.close();
      sink = null;
      t.total = t.received;
      t.status = 'done';
      _notify();
      await _save();
    } catch (_) {
      try {
        await sink?.close();
      } catch (_) {}
      try {
        if (f != null && f.existsSync()) f.deleteSync();
      } catch (_) {}
      t.status = t.cancel ? 'canceled' : 'failed';
      _notify();
    } finally {
      client.close();
    }
  }

  static Future<void> remove(DlTask t) async {
    if (t.status == 'downloading') {
      t.cancel = true;
      t.client?.close();
    }
    try {
      if (t.path.isNotEmpty && File(t.path).existsSync()) File(t.path).deleteSync();
    } catch (_) {}
    tasks.value = tasks.value.where((e) => e.id != t.id).toList();
    await _save();
  }
}

String _fmt(int b) {
  if (b <= 0) return '';
  final mb = b / (1024 * 1024);
  return mb >= 1024 ? '${(mb / 1024).toStringAsFixed(2)} GB' : '${mb.toStringAsFixed(1)} MB';
}

class DownloadsScreen extends StatelessWidget {
  const DownloadsScreen({super.key});

  Future<void> _open(BuildContext c, DlTask t) async {
    final r = await OpenFilex.open(t.path);
    if (r.type != ResultType.done && c.mounted) {
      ScaffoldMessenger.of(c).showSnackBar(const SnackBar(content: Text('No video player found to open this file.')));
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Downloads', style: TextStyle(fontWeight: FontWeight.w800))),
      body: ValueListenableBuilder<List<DlTask>>(
        valueListenable: Downloads.tasks,
        builder: (_, list, __) {
          if (list.isEmpty) {
            return const EmptyState(
              icon: Icons.download_for_offline_outlined,
              title: 'No downloads yet',
              subtitle: 'Your downloaded videos will appear here.',
            );
          }
          return ListView.separated(
            padding: const EdgeInsets.all(16),
            itemCount: list.length,
            separatorBuilder: (_, __) => const SizedBox(height: 10),
            itemBuilder: (c, i) {
              final t = list[i];
              final busy = t.status == 'downloading';
              final done = t.status == 'done';
              final sub = busy
                  ? '${t.quality.isEmpty ? '' : '${t.quality} · '}${_fmt(t.received)}${t.total > 0 ? ' / ${_fmt(t.total)}' : ''}'
                  : done
                      ? '${t.quality.isEmpty ? '' : '${t.quality} · '}${_fmt(t.total)}'
                      : (t.status == 'failed' ? 'Failed' : 'Canceled');
              return Container(
                padding: const EdgeInsets.all(12),
                decoration: BoxDecoration(color: kCard, borderRadius: BorderRadius.circular(16)),
                child: Row(
                  children: [
                    Icon(done ? Icons.movie_rounded : (busy ? Icons.downloading_rounded : Icons.error_outline_rounded),
                        color: done ? Colors.greenAccent : (busy ? Colors.white70 : Colors.orangeAccent), size: 30),
                    const SizedBox(width: 12),
                    Expanded(
                      child: Column(
                        crossAxisAlignment: CrossAxisAlignment.start,
                        children: [
                          Text(t.title, maxLines: 1, overflow: TextOverflow.ellipsis, style: const TextStyle(fontWeight: FontWeight.w700)),
                          const SizedBox(height: 3),
                          Text(sub, style: const TextStyle(fontSize: 12, color: Colors.white54)),
                          if (busy) ...[
                            const SizedBox(height: 8),
                            LinearProgressIndicator(
                              value: t.total > 0 ? t.progress : null,
                              color: kRed,
                              backgroundColor: Colors.white10,
                              minHeight: 4,
                            ),
                          ],
                        ],
                      ),
                    ),
                    if (done) IconButton(icon: const Icon(Icons.play_circle_fill_rounded, color: kRedLight, size: 30), onPressed: () => _open(c, t)),
                    IconButton(
                      icon: Icon(busy ? Icons.close_rounded : Icons.delete_outline_rounded, color: Colors.white54),
                      onPressed: () => Downloads.remove(t),
                    ),
                  ],
                ),
              );
            },
          );
        },
      ),
    );
  }
}

class _BootState extends State<Boot> {
  int tab = 0;
  Timer? _t;

  @override
  void initState() {
    super.initState();
    final u = FirebaseAuth.instance.currentUser;
    if (u != null) UserData.start(u.uid);
    Store.start();
    Api.load();
    Downloads.load();

        ],
      ),
    );
  }
}

      ));
    }
    final movies = all.where((e) => e['type'] != 'tv').toList();
    final series = all.where((e) => e['type'] == 'tv').toList();
    final cont = UserData.sorted(hist, 'watchedAt').take(12).toList();
    final counts = <String, int>{};
    for (final e in all) {
      final g = e['genres'];
      if (g is List) {

    }

  }

class _SlideShowState extends State<SlideShow> {

  Timer? _t;

  List<Item> get _slides => Store.slides.value.where((a) => a['active'] != false).toList();

  @override
  void initState() {
    super.initState();
    _t = Timer.periodic(const Duration(seconds: 5), (_) {
      final n = _slides.length;
      if (n < 2 || !_pc.hasClients) return;
      _pc.animateToPage((_i + 1) % n, duration: const Duration(milliseconds: 400), curve: Curves.easeOut);
    });
  }

  @override
  void dispose() {
    _t?.cancel();
    _pc.dispose();
    super.dispose();
  }

  void _open(Item s) {
    final id = '${s['contentId'] ?? ''}';
    if (id.isNotEmpty) {
      for (final e in widget.all) {
        if ('${e['id']}' == id) {
          openItem(context, e);
          return;
        }
      }
    }
    openLink('${s['link'] ?? ''}');
  }

  @override

    return ValueListenableBuilder<List<Item>>(
      valueListenable: Store.slides,
      builder: (_, __, ___) {
        final l = _slides;
        if (l.isEmpty) return const SizedBox.shrink();

          height: 200,
          child: PageView.builder(
            controller: _pc,
            onPageChanged: (i) => _i = i,
            itemCount: l.length,
            itemBuilder: (_, i) {
              final s = l[i];

typedef Src = Map<String, String>;

class DetailScreen extends StatefulWidget {
  final Item item;
  const DetailScreen({super.key, required this.item});

  @override
  State<DetailScreen> createState() => _DetailScreenState();
}

class _DetailScreenState extends State<DetailScreen> {
  Item get it => widget.item;
  Map<String, dynamic>? omdb;
  int season = 1, episode = 1, seasonCount = 1, epCount = 1, serverIdx = 0;
  String quality = 'Auto';
  List<String> epNames = [];
  bool playing = false;

  bool get isTv => it['type'] == 'tv';
  String get tmdbId => '${it['tmdbId'] ?? ''}';

  @override
  void initState() {
    super.initState();
    final h = UserData.history.value['${it['id']}'];
    if (h != null && isTv) {
      season = (h['season'] as num?)?.toInt() ?? 1;
      episode = (h['episode'] as num?)?.toInt() ?? 1;
    }
    _loadExtra();
  }

  Future<void> _loadExtra() async {
    if (tmdbId.isEmpty) return;
    String imdb = '';
    if (isTv) {
      final d = await Api.tmdb('tv/$tmdbId', {'append_to_response': 'external_ids'});
      if (d == null) return;
      imdb = '${d['external_ids']?['imdb_id'] ?? ''}';
      final n = (d['number_of_seasons'] as num?)?.toInt() ?? 1;
      if (mounted) setState(() => seasonCount = n < 1 ? 1 : n);
      _loadEpisodes();
    } else {
      final d = await Api.tmdb('movie/$tmdbId');
      imdb = '${d?['imdb_id'] ?? ''}';
    }
    final o = await Api.omdb(imdb);
    if (mounted && o != null) setState(() => omdb = o);
  }

  Future<void> _loadEpisodes() async {
    final d = await Api.tmdb('tv/$tmdbId/season/$season');
    final eps = d?['episodes'];
    if (eps is List && mounted) {
      setState(() {
        epCount = eps.isEmpty ? 1 : eps.length;
        epNames = [for (final e in eps) '${e is Map ? (e['name'] ?? '') : ''}'];
        if (episode > epCount) episode = 1;
      });
    }
  }

  String _fill(String tpl) => tpl
      .replaceAll('{tmdb_id}', tmdbId)
      .replaceAll('{season}', '$season')
      .replaceAll('{episode}', '$episode');

  // ---------- quality helpers ----------
  String _guessQuality(String explicit, String name) {
    if (explicit.trim().isNotEmpty) return explicit.trim();
    final m = RegExp(r'(2160p|4k|1440p|1080p|720p|480p|360p|240p|cam|hd)', caseSensitive: false).firstMatch(name);
    if (m == null) return '';
    final v = m.group(0)!.toLowerCase();
    return (v == '4k' || v == 'cam' || v == 'hd') ? v.toUpperCase() : v;
  }

  int _qv(String q) => q == '4K' ? 2160 : (int.tryParse(q.replaceAll('p', '')) ?? 0);

  List<String> _qualities(List<Src> l) {
    final s = <String>{
      for (final e in l)
        if ((e['quality'] ?? '').isNotEmpty) e['quality']!
    };
    return s.toList()..sort((a, b) => _qv(b).compareTo(_qv(a)));
  }

  // ---------- play sources ----------
  List<Src> _playServers() {
    final l = <Src>[];
    if (tmdbId.isNotEmpty) {
      for (final s in Store.servers.value.where((s) => s['active'] != false)) {
        final tpl = '${(isTv ? s['tvUrl'] : s['movieUrl']) ?? ''}';
        if (tpl.isEmpty) continue;
        final name = '${s['name'] ?? 'Server'}';
        l.add({'name': name, 'url': _fill(tpl), 'quality': _guessQuality('${s['quality'] ?? ''}', name)});
      }
    }
    final own = it['servers'];
    if (own is List) {
      for (final m in own) {
        if (m is Map && '${m['url'] ?? ''}'.isNotEmpty) {
          final name = '${m['name'] ?? 'Server'}';
          l.add({'name': name, 'url': _fill('${m['url']}'), 'quality': _guessQuality('${m['quality'] ?? ''}', name)});
        }
      }
    }
    return l;
  }

  // ---------- download sources ----------
  List<Src> _dlSources() {
    final l = <Src>[];
    if (tmdbId.isNotEmpty) {
      for (final s in Store.servers.value.where((s) => s['active'] != false)) {
        final tpl = '${(isTv ? s['dlTvUrl'] : s['dlMovieUrl']) ?? ''}';
        if (tpl.isEmpty) continue;
        final name = '${s['name'] ?? 'Download'}';
        l.add({'name': name, 'url': _fill(tpl), 'quality': _guessQuality('${s['quality'] ?? ''}', name)});
      }
    }
    final dl = it['downloads'];
    if (dl is List) {
      for (final m in dl) {
        if (m is Map && '${m['url'] ?? ''}'.isNotEmpty) {
          final name = '${m['name'] ?? 'Download'}';
          l.add({'name': name, 'url': _fill('${m['url']}'), 'quality': _guessQuality('${m['quality'] ?? ''}', name)});
        }
      }
    }
    return l;
  }

  String get _label => isTv ? '${it['title']} S${season}E$episode' : '${it['title']}';

  void _startDownload(Src s) {
    final url = s['url']!;
    if (Downloads.isDirect(url)) {
      Downloads.start(_label, s['quality'] ?? '', url);
      ScaffoldMessenger.of(context).showSnackBar(SnackBar(
        content: const Text('Download started'),
        action: SnackBarAction(
          label: 'VIEW',
          onPressed: () => Navigator.push(context, MaterialPageRoute(builder: (_) => const DownloadsScreen())),
        ),
      ));
    } else {
      Navigator.push(context, MaterialPageRoute(builder: (_) => WebPage(url: url, title: _label)));
    }
  }

  void _showDownloadOptions() {
    final direct = _dlSources();
    final src = direct.isNotEmpty ? direct : _playServers();
    if (src.isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('No download source available.')));
      return;
    }
    String q = 'All';
    showModalBottomSheet(
      context: context,
      isScrollControlled: true,
      backgroundColor: kCard,
      shape: const RoundedRectangleBorder(borderRadius: BorderRadius.vertical(top: Radius.circular(20))),
      builder: (ctx) => StatefulBuilder(builder: (ctx, setS) {
        final qs = _qualities(src);
        final list = q == 'All' ? src : src.where((s) => s['quality'] == q).toList();
        return SafeArea(
          child: Padding(
            padding: const EdgeInsets.all(20),
            child: Column(
              mainAxisSize: MainAxisSize.min,
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                const Text('Download', style: TextStyle(fontSize: 18, fontWeight: FontWeight.w800)),
                const SizedBox(height: 2),
                Text(_label, style: const TextStyle(color: Colors.white54, fontSize: 12.5)),
                if (qs.isNotEmpty) ...[
                  const SizedBox(height: 14),
                  const Text('Quality', style: TextStyle(fontWeight: FontWeight.w700)),
                  const SizedBox(height: 8),
                  Wrap(
                    spacing: 8,
                    children: [

  Widget _seasonRow() {

      height: 40,
      child: ListView.separated(
        scrollDirection: Axis.horizontal,
        physics: const BouncingScrollPhysics(),
        itemCount: seasonCount,
        separatorBuilder: (_, __) => const SizedBox(width: 8),
        itemBuilder: (_, i) {
          final s = i + 1;
          final sel = s == season;
          return GestureDetector(
            onTap: () {
              if (sel) return;
              setState(() {
                season = s;
                episode = 1;
                epNames = [];
              });
              _loadEpisodes();
              if (playing) UserData.addHistory(it, season: season, episode: episode);
            },
            child: AnimatedContainer(
              duration: const Duration(milliseconds: 220),
              padding: const EdgeInsets.symmetric(horizontal: 18),
              alignment: Alignment.center,
              decoration: BoxDecoration(
                gradient: sel ? kRedGradient : null,
                color: sel ? null : kCard,
                borderRadius: BorderRadius.circular(20),
                border: Border.all(color: sel ? Colors.transparent : Colors.white10),
              ),
              child: Text('Season $s', style: TextStyle(fontWeight: FontWeight.w700, color: sel ? Colors.white : Colors.white70)),
            ),
          );
        },
      ),
    );
  }

  void _pickEpisode(int e) {
    setState(() => episode = e);
    if (playing) UserData.addHistory(it, season: season, episode: episode);
  }

  Widget _episodeRow() {
    return Column(
      crossAxisAlignment: CrossAxisAlignment.start,
      children: [
        Text('Episodes · $epCount', style: const TextStyle(fontWeight: FontWeight.w800, fontSize: 15)),
        const SizedBox(height: 8),
        SizedBox(
          height: 74,
          child: ListView.separated(
            scrollDirection: Axis.horizontal,
            physics: const BouncingScrollPhysics(),
            itemCount: epCount,
            separatorBuilder: (_, __) => const SizedBox(width: 10),
            itemBuilder: (_, i) {
              final e = i + 1;
              final sel = e == episode;
              final name = i < epNames.length ? epNames[i] : '';
              return GestureDetector(
                onTap: () => _pickEpisode(e),
                child: AnimatedContainer(
                  duration: const Duration(milliseconds: 220),
                  width: 132,
                  padding: const EdgeInsets.all(12),
                  decoration: BoxDecoration(
                    color: sel ? kRed.withOpacity(0.16) : kCard,
                    borderRadius: BorderRadius.circular(16),
                    border: Border.all(color: sel ? kRed : Colors.white10, width: sel ? 1.6 : 1),
                  ),
                  child: Column(
                    crossAxisAlignment: CrossAxisAlignment.start,
                    mainAxisAlignment: MainAxisAlignment.center,
                    children: [
                      Row(children: [
                        if (sel)
                          const Padding(
                            padding: EdgeInsets.only(right: 4),
                            child: Icon(Icons.play_arrow_rounded, size: 18, color: kRedLight),
                          ),
                        Text('EP $e', style: TextStyle(fontWeight: FontWeight.w800, color: sel ? kRedLight : Colors.white, fontSize: 14)),
                      ]),
                      if (name.isNotEmpty) ...[
                        const SizedBox(height: 3),
                        Text(name, maxLines: 1, overflow: TextOverflow.ellipsis, style: const TextStyle(fontSize: 11.5, color: Colors.white54)),
                      ],
                    ],
                  ),
                ),
              );
            },
          ),
        ),
      ],
    );
  }

  List<Item> get related {
    final g = it['genres'];
    final mine = g is List ? g.map((e) => '$e').toSet() : <String>{};
    if (mine.isEmpty) return [];
    int score(Item e) {
      final x = e['genres'];
      return x is List ? x.where((a) => mine.contains('$a')).length : 0;
    }

    final l = Store.content.value.where((e) => e['id'] != it['id'] && score(e) > 0).toList();
    l.sort((a, b) => score(b).compareTo(score(a)));
    return l.take(12).toList();
  }

  @override

    final all = _playServers();
    final qs = _qualities(all);
    final qSel = qs.contains(quality) ? quality : 'Auto';
    final sv = qSel == 'Auto' ? all : all.where((s) => s['quality'] == qSel).toList();
    final idx = sv.isEmpty ? 0 : (serverIdx >= sv.length ? 0 : serverIdx);
    final url = sv.isEmpty ? '' : sv[idx]['url']!;
    final g = it['genres'];
    final genres = g is List ? g.map((e) => '$e').toList() : <String>[];
    final backdrop = '${it['backdrop'] ?? ''}'.isNotEmpty ? '${it['backdrop']}' : '${it['poster'] ?? ''}';

              aspectRatio: 16 / 9,
              child: Stack(
                fit: StackFit.expand,
                children: [

                  else ...[
                    Img(backdrop),
                    const DecoratedBox(
                      decoration: BoxDecoration(
                        gradient: LinearGradient(
                          begin: Alignment.topCenter,
                          end: Alignment.bottomCenter,

                          stops: [0, 0.4, 1],
                        ),
                      ),
                    ),
                    Center(
                      child: GestureDetector(
                        onTap: _play,
                        child: Container(

                          height: 68,
                          decoration: BoxDecoration(
                            gradient: kRedGradient,
                            shape: BoxShape.circle,
                            border: Border.all(color: Colors.white24, width: 2),
                            boxShadow: [BoxShadow(color: kRed.withOpacity(0.55), blurRadius: 28)],

                          child: const Icon(Icons.play_arrow_rounded, size: 42),

                      ),

  void _play() {
    if (_playServers().isEmpty) {
      ScaffoldMessenger.of(context).showSnackBar(const SnackBar(content: Text('No server available for this title yet.')));
      return;
    }
    setState(() => playing = true);
    UserData.addHistory(it, season: season, episode: episode);
  }

class PlayerView extends StatefulWidget {
  final String url;
  const PlayerView({super.key, required this.url});

  @override
  State<PlayerView> createState() => _PlayerViewState();
}

class _PlayerViewState extends State<PlayerView> {
  late final WebViewController c;
  bool loading = true;
  bool loaded = false;

  String _root(String h) {
    final p = h.split('.');
    return p.length >= 2 ? p.sublist(p.length - 2).join('.') : h;
  }

  @override
  void initState() {
    super.initState();
    final base = _root(Uri.tryParse(widget.url)?.host ?? '');
    c = WebViewController()
      ..setJavaScriptMode(JavaScriptMode.unrestricted)
      ..setBackgroundColor(Colors.black)
      ..setNavigationDelegate(NavigationDelegate(
        onPageFinished: (_) {
          loaded = true;
          if (mounted) setState(() => loading = false);
        },
        onNavigationRequest: (r) {
          if (!loaded || !r.isMainFrame) return NavigationDecision.navigate;
          final u = Uri.tryParse(r.url);
          if (u == null) return NavigationDecision.prevent;
          if (u.scheme == 'about' || u.scheme == 'data' || _root(u.host) == base) {
            return NavigationDecision.navigate;
          }
          return NavigationDecision.prevent;
        },
      ));
    c.loadRequest(Uri.parse(widget.url));
  }

  @override
  Widget build(BuildContext context) {
    return Stack(
      fit: StackFit.expand,
      children: [
        WebViewWidget(controller: c),
        if (loading) const Center(child: CircularProgressIndicator(color: kRed)),
      ],
    );
  }
}
